﻿using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.SceneManagement;
using DG.Tweening;

public class GlobalGameManager : MonoBehaviour
{
    public static GlobalGameManager Instance;

    [Header("Transition Settings")]
    [SerializeField] private float countdownDelay = 3f;

    // [VI] Thêm biến thời gian chờ sau khi hiện Banner (2s)
    // [EN] Added wait time after showing the Banner (2s)
    [SerializeField] private float bannerDisplayDuration = 2f;

    // [VI] Đổi từ Prefab thành GameObject có sẵn trong Scene
    // [EN] Changed from Prefab to an existing GameObject in the Scene
    [SerializeField] private GameObject transitionBanner;
    [SerializeField] private string animationStateName = "Play";

    [HideInInspector]
    public GameLevelData currentLevelData;
    [HideInInspector]
    public List<InputDevice> joinedPlayerDevices = new List<InputDevice>();
    private bool isLoadingScene = false;

    // [VI] Biến lưu trữ Coroutine đếm ngược để có thể reset
    // [EN] Variable to store the countdown Coroutine so it can be reset
    private Coroutine countdownCoroutine;

    [Header("Black Screen Transition")]
    [SerializeField] private RectTransform blackScreen; // [VI] Kéo BlackScreen vào đây / [EN] Drag BlackScreen here
    [SerializeField] private float transitionDuration = 0.5f;

    private void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);

            // [VI] Đăng ký sự kiện khi load xong Scene mới
            // [EN] Subscribe to the event when a new Scene finishes loading
            SceneManager.sceneLoaded += OnSceneLoaded;
        }
        else
        {
            Destroy(gameObject);
        }
    }

    private void Start()
    {
        // [VI] Đảm bảo màn hình đen nằm ngoài rìa trái lúc mới bắt đầu game
        // [EN] Ensure the black screen is off the left edge when the game starts
        if (blackScreen != null)
        {
            blackScreen.anchoredPosition = new Vector2(-Screen.width * 2, 0);
        }
    }

    public void StartGameChecker()
    {
        if (currentLevelData == null || isLoadingScene) return;

        // [VI] Đổi thành >= minPlayers để kích hoạt đếm ngược
        // [EN] Changed to >= minPlayers to trigger the countdown
        if (joinedPlayerDevices.Count >= currentLevelData.minPlayers)
        {
            // [VI] Nếu đang đếm ngược mà có người vào, DỪNG đếm ngược cũ để reset về 0
            // [EN] If already counting down and someone joins, STOP the old countdown to reset to 0
            if (countdownCoroutine != null)
            {
                StopCoroutine(countdownCoroutine);
            }

            Debug.Log($"Đạt {joinedPlayerDevices.Count} người! Chờ {countdownDelay}s xem có ai vào thêm không...");

            // [VI] Bắt đầu lượt đếm ngược mới
            // [EN] Start a new countdown cycle
            countdownCoroutine = StartCoroutine(CountdownAndLoadRoutine());
        }
    }

    private IEnumerator CountdownAndLoadRoutine()
    {
        // 1. [VI] Chờ xem có người chơi mới không (thời gian này sẽ bị reset nếu StartGameChecker được gọi lại)
        // 1. [EN] Wait to see if new players join (this time resets if StartGameChecker is called again)
        yield return new WaitForSeconds(countdownDelay);

        // 2. [VI] Hết thời gian chờ, KHÓA hàm lại không cho người khác vào làm reset nữa
        // 2. [EN] Countdown finished, LOCK the function so new joins won't reset it anymore
        isLoadingScene = true;

        // 3. [VI] Hiện Banner
        // 3. [EN] Show Banner
        if (transitionBanner != null)
        {
            transitionBanner.SetActive(true);
            Animator anim = transitionBanner.GetComponentInChildren<Animator>();

            if (anim != null && !string.IsNullOrEmpty(animationStateName))
            {
                anim.Update(0);
                anim.Play(animationStateName, -1, 0f);
            }
        }

        // 4. [VI] Chờ Banner chạy (2s)
        // 4. [EN] Wait for Banner to play (2s)
        yield return new WaitForSeconds(bannerDisplayDuration);

        // 5. [VI] Kéo màn hình đen và Load Scene
        // 5. [EN] Pull black screen and Load Scene
        if (blackScreen != null)
        {
            blackScreen.anchoredPosition = new Vector2(-Screen.width * 2, 0);
            blackScreen.DOAnchorPosX(0, transitionDuration).SetEase(Ease.InOutSine);

            yield return new WaitForSeconds(transitionDuration);
        }

        SceneManager.LoadScene(currentLevelData.sceneName);
    }

    private void OnSceneLoaded(Scene scene, LoadSceneMode mode)
    {
        if (blackScreen != null)
        {
            // [VI] Kéo màn đen từ Giữa (0) sang Phải (Screen.width) để mở màn
            // [EN] Pull black screen from Center (0) to Right (Screen.width) to reveal
            blackScreen.anchoredPosition = new Vector2(0, 0);
            blackScreen.DOAnchorPosX(Screen.width * 2, transitionDuration).SetEase(Ease.InOutSine);
        }
    }

    private void OnDestroy()
    {
        // [VI] Hủy đăng ký sự kiện để tránh lỗi memory leak
        // [EN] Unsubscribe from the event to prevent memory leaks
        SceneManager.sceneLoaded -= OnSceneLoaded;
    }

    public void AddPlayer(PlayerInput playerInput)
    {
        if (playerInput.devices.Count == 0)
        {
            Debug.LogWarning("Player joined but no device was found!");
            return;
        }

        InputDevice newDevice = playerInput.devices[0];

        if (!joinedPlayerDevices.Contains(newDevice))
        {
            joinedPlayerDevices.Add(newDevice);
            Debug.Log($"Player {joinedPlayerDevices.Count} Joined with {newDevice.name}");
        }

        // [VI] Gọi hàm check mỗi khi có người mới vào để chạy hoặc reset đếm ngược
        // [EN] Call the check function every time someone joins to start or reset the countdown
        StartGameChecker();
    }

    public int GetPlayerCount()
    {
        return joinedPlayerDevices.Count;
    }
}

﻿using UnityEngine;
using UnityEngine.InputSystem;
using DG.Tweening;
using TMPro;

public class LobbyManager : MonoBehaviour
{
    [Header("UI References")]
    [SerializeField] private Transform avatarLayoutGroup;     // [VI] Panel chứa Layout / [EN] Panel containing Layout
    [SerializeField] private GameObject avatarPrefab;         // [VI] Prefab của Avatar / [EN] Avatar Prefab
    [SerializeField] private TextMeshProUGUI playerCountText; // [VI] Text hiển thị Player / Max / [EN] Text displaying Player / Max
    [SerializeField] private RectTransform teamSetBanner;

    [HideInInspector] public GameLevelData currentLevelData;
    public int currentPlayerCount = 0;

    private void Start()
    {
        if (teamSetBanner != null)
        {
            teamSetBanner.anchoredPosition = new Vector2(-1500, teamSetBanner.anchoredPosition.y);
        }
        UpdatePlayerCountText();
    }

    public void OnPlayerJoined(PlayerInput playerInput)
    {
        // [VI] 1. Kiểm tra an toàn dữ liệu / [EN] 1. Data safety check
        if (currentLevelData == null || currentPlayerCount >= currentLevelData.maxPlayers)
        {
            Debug.Log("Phòng đầy hoặc chưa chọn Map!");
            Destroy(playerInput.gameObject);
            return;
        }

        // [VI] 2. Kiểm tra loại Input / [EN] 2. Check Input type
        if (!IsInputValid(playerInput))
        {
            Debug.LogWarning($"Từ chối: Sai thiết bị. Yêu cầu: {currentLevelData.inputType}");
            Destroy(playerInput.gameObject);
            return;
        }

        // [VI] 3. Đăng ký người chơi với GlobalGameManager / [EN] 3. Register player to GlobalGameManager
        GlobalGameManager.Instance.AddPlayer(playerInput);
        currentPlayerCount = GlobalGameManager.Instance.GetPlayerCount();

        // [VI] 4. Debug loại Input và số thứ tự Player / [EN] 4. Debug Input type and Player number
        InputDevice device = playerInput.devices[0];
        Debug.Log($"Player {currentPlayerCount} joined. Device: {device.name} (Layout: {device.layout})");

        // [VI] 5. Cập nhật UI / [EN] 5. Update UI
        UpdatePlayerCountText();
        CreateAvatar();

        // [VI] 6. Kiểm tra đủ người để hiện Banner / [EN] 6. Check if full to show Banner
        if (currentPlayerCount == currentLevelData.maxPlayers)
        {
            ShowTeamSetBanner();
        }
    }

    // [VI] Hàm tạo Avatar bằng Prefab / [EN] Function to create Avatar using Prefab
    private void CreateAvatar()
    {
        if (avatarPrefab == null || avatarLayoutGroup == null)
        {
            Debug.LogError("Lỗi: Thiếu Avatar Prefab hoặc Layout Group trong Inspector!");
            return;
        }

        Instantiate(avatarPrefab, avatarLayoutGroup);
    }

    // [VI] Hàm cập nhật Text Player/Max / [EN] Function to update Player/Max Text
    private void UpdatePlayerCountText()
    {
        if (playerCountText != null && currentLevelData != null)
        {
            playerCountText.text = $"{currentPlayerCount} / {currentLevelData.maxPlayers}";
        }
    }

    private void ShowTeamSetBanner()
    {
        if (teamSetBanner != null)
        {
            teamSetBanner.DOAnchorPosX(0, 0.6f).SetEase(Ease.OutBack);
        }
    }

    private bool IsInputValid(PlayerInput playerInput)
    {
        if (playerInput.devices.Count == 0) return false;

        InputDevice device = playerInput.devices[0];
        SupportedInput requiredInput = currentLevelData.inputType;

        if (requiredInput == SupportedInput.Any) return true;
        if (requiredInput == SupportedInput.GamepadOnly && device is Gamepad) return true;
        if (requiredInput == SupportedInput.KeyboardOnly && device is Keyboard) return true;

        return false;
    }
}

﻿using UnityEngine;

public class MapSelectionManager : MonoBehaviour
{
    [Header("UI Panels")]
    [SerializeField] private GameObject mapSelectionPanel;
    [SerializeField] private GameObject registerPanel;

    [Header("Managers")]
    [SerializeField] private LobbyManager registerManager;

    // [VI] Hàm chính cho Button gọi / [EN] Main function for Button to call
    public void OnMapSelected(GameLevelData selectedLevelData)
    {
        SetLevelData(selectedLevelData);
        TransitionToRegister();
    }

    // [VI] 1. Hàm gửi data về GameManager / [EN] 1. Function to send data to GameManager
    private void SetLevelData(GameLevelData data)
    {
        GlobalGameManager.Instance.currentLevelData = data;

        if (registerManager != null)
        {
            registerManager.currentLevelData = data;
        }
    }

    // [VI] 2. Hàm tắt/mở Panel và Manager / [EN] 2. Function to toggle Panels and Manager
    private void TransitionToRegister()
    {
        mapSelectionPanel.SetActive(false);
        registerPanel.SetActive(true);

        if (registerManager != null)
        {
            // [VI] Bật GameObject chứa script lên trước / [EN] Enable the GameObject containing the script first
            registerManager.gameObject.SetActive(true);
            // [VI] Bật component / [EN] Enable the component
            registerManager.enabled = true;
        }
    }
}

﻿using UnityEngine;

public class VFXManager : MonoBehaviour
{
    // [VI] Tạo Singleton để gọi từ bất kỳ script nào khác
    // [EN] Create Singleton for global access from any other script
    public static VFXManager Instance { get; private set; }

    [Header("VFX Prefabs")]
    public GameObject killEffectPrefab;
    public GameObject collideEffectPrefab;
    public GameObject lightningEffectPrefab;
    public GameObject spawnEffectPrefab;

    private void Awake()
    {
        // [VI] Đảm bảo chỉ có 1 VFXManager tồn tại trong Scene
        // [EN] Ensure only one VFXManager exists in the Scene
        if (Instance != null && Instance != this)
        {
            Destroy(this.gameObject);
            return;
        }
        Instance = this;
    }

    // --- [VI] CÁC HÀM GỌI EFFECT / [EN] EFFECT SPAWN FUNCTIONS ---

    public void PlayKillEffect(Vector3 position, float destroyDelay = 2f)
    {
        SpawnAndDestroy(killEffectPrefab, position, destroyDelay);
    }

    public void PlayCollideEffect(Vector3 position, float destroyDelay = 2f)
    {
        SpawnAndDestroy(collideEffectPrefab, position, destroyDelay);
    }

    public void PlayLightningEffect(Vector3 position, float destroyDelay = 2f)
    {
        SpawnAndDestroy(lightningEffectPrefab, position, destroyDelay);
    }

    public void PlaySpawnEffect(Vector3 position, float destroyDelay = 2f)
    {
        SpawnAndDestroy(spawnEffectPrefab, position, destroyDelay);
    }

    // [VI] Hàm nội bộ xử lý việc sinh ra và tự động xóa
    // [EN] Internal helper to handle instantiation and auto-destruction
    private void SpawnAndDestroy(GameObject prefab, Vector3 spawnPosition, float destroyTime)
    {
        if (prefab == null)
        {
            Debug.LogWarning("VFXManager: Prefab is missing! / Thiếu Prefab!");
            return;
        }

        // [VI] Sinh ra effect / [EN] Spawn the effect
        GameObject spawnedEffect = Instantiate(prefab, spawnPosition, Quaternion.identity);

        // [VI] Tự động xóa sau thời gian chỉ định / [EN] Auto-destroy after set time
        Destroy(spawnedEffect, destroyTime);
    }
}

        /// <summary>Tweens a Rigidbody's position to the given value, while also applying a jump effect along the Y axis.
        /// Returns a Sequence instead of a Tweener.
        /// Also stores the Rigidbody as the tween's target so it can be used for filtered operations</summary>
        /// <param name="endValue">The end value to reach</param>
        /// <param name="jumpPower">Power of the jump (the max height of the jump is represented by this plus the final Y offset)</param>
        /// <param name="numJumps">Total number of jumps</param>
        /// <param name="duration">The duration of the tween</param>
        /// <param name="snapping">If TRUE the tween will smoothly snap all values to integers</param>
        public static Sequence DOJump(this Rigidbody target, Vector3 endValue, float jumpPower, int numJumps, float duration, bool snapping = false)
        {
            if (numJumps < 1) numJumps = 1;
            float startPosY = 0;
            float offsetY = -1;
            bool offsetYSet = false;
            Sequence s = DOTween.Sequence();
            Tween yTween = DOTween.To(() => target.position, target.MovePosition, new Vector3(0, jumpPower, 0), duration / (numJumps * 2))
                .SetOptions(AxisConstraint.Y, snapping).SetEase(Ease.OutQuad).SetRelative()
                .SetLoops(numJumps * 2, LoopType.Yoyo)
                .OnStart(() => startPosY = target.position.y);
            s.Append(DOTween.To(() => target.position, target.MovePosition, new Vector3(endValue.x, 0, 0), duration)
                    .SetOptions(AxisConstraint.X, snapping).SetEase(Ease.Linear)
                ).Join(DOTween.To(() => target.position, target.MovePosition, new Vector3(0, 0, endValue.z), duration)
                    .SetOptions(AxisConstraint.Z, snapping).SetEase(Ease.Linear)
                ).Join(yTween)
                .SetTarget(target).SetEase(DOTween.defaultEaseType);
            yTween.OnUpdate(() => {
                if (!offsetYSet) {
                    offsetYSet = true;
                    offsetY = s.isRelative ? endValue.y : endValue.y - startPosY;
                }
                Vector3 pos = target.position;
                pos.y += DOVirtual.EasedValue(0, offsetY, yTween.ElapsedPercentage(), Ease.OutQuad);
                target.MovePosition(pos);
            });
            return s;
        }

                .SetTarget(target).SetUpdate(UpdateType.Fixed);

            t.plugOptions.isRigidbody2D = true;
            t.plugOptions.mode = pathMode;
            t.plugOptions.useLocalPosition = true;
            return t;
        }
        // Used by path editor when creating the actual tween, so it can pass a pre-compiled path
        internal static TweenerCore<Vector3, Path, PathOptions> DOPath(

        /// <param name="endValue">The end value to reach</param><param name="duration">The duration of the tween</param>
        public static TweenerCore<Color, Color, ColorOptions> DOFade(this SpriteRenderer target, float endValue, float duration)

        /// <summary>Tweens a SpriteRenderer's color using the given gradient
        /// (NOTE 1: only uses the colors of the gradient, not the alphas - NOTE 2: creates a Sequence, not a Tweener).
        /// Also stores the image as the tween's target so it can be used for filtered operations</summary>
        /// <param name="gradient">The gradient to use</param><param name="duration">The duration of the tween</param>
        public static Sequence DOGradientColor(this SpriteRenderer target, Gradient gradient, float duration)
        {
            Sequence s = DOTween.Sequence();
            GradientColorKey[] colors = gradient.colorKeys;
            int len = colors.Length;
            for (int i = 0; i < len; ++i) {
                GradientColorKey c = colors[i];
                if (i == 0 && c.time <= 0) {
                    target.color = c.color;
                    continue;
                }
                float colorDuration = i == len - 1
                    ? duration - s.Duration(false) // Verifies that total duration is correct
                    : duration * (i == 0 ? c.time : c.time - colors[i - 1].time);
                s.Append(target.DOColor(c.color, colorDuration).SetEase(Ease.Linear));
            }
            s.SetTarget(target);
            return s;
        }

        #region Blendables

        #region SpriteRenderer

        /// <summary>Tweens a SpriteRenderer's color to the given value,
        /// in a way that allows other DOBlendableColor tweens to work together on the same target,
        /// instead than fight each other as multiple DOColor would do.
        /// Also stores the SpriteRenderer as the tween's target so it can be used for filtered operations</summary>
        /// <param name="endValue">The value to tween to</param><param name="duration">The duration of the tween</param>
        public static Tweener DOBlendableColor(this SpriteRenderer target, Color endValue, float duration)
        {
            endValue = endValue - target.color;
            Color to = new Color(0, 0, 0, 0);
            return DOTween.To(() => to, x => {
                    Color diff = x - to;
                    to = x;
                    target.color += diff;
                }, endValue, duration)
                .Blendable().SetTarget(target);
        }

        #endregion

        #endregion

        #endregion
	}
}

        /// <summary>Tweens a RectTransform's anchoredPosition to the given value, while also applying a jump effect along the Y axis.
        /// Returns a Sequence instead of a Tweener.
        /// Also stores the RectTransform as the tween's target so it can be used for filtered operations</summary>
        /// <param name="endValue">The end value to reach</param>
        /// <param name="jumpPower">Power of the jump (the max height of the jump is represented by this plus the final Y offset)</param>
        /// <param name="numJumps">Total number of jumps</param>
        /// <param name="duration">The duration of the tween</param>
        /// <param name="snapping">If TRUE the tween will smoothly snap all values to integers</param>
        public static Sequence DOJumpAnchorPos(this RectTransform target, Vector2 endValue, float jumpPower, int numJumps, float duration, bool snapping = false)
        {
            if (numJumps < 1) numJumps = 1;
            float startPosY = 0;
            float offsetY = -1;
            bool offsetYSet = false;

            // Separate Y Tween so we can elaborate elapsedPercentage on that insted of on the Sequence
            // (in case users add a delay or other elements to the Sequence)
            Sequence s = DOTween.Sequence();
            Tween yTween = DOTween.To(() => target.anchoredPosition, x => target.anchoredPosition = x, new Vector2(0, jumpPower), duration / (numJumps * 2))
                .SetOptions(AxisConstraint.Y, snapping).SetEase(Ease.OutQuad).SetRelative()
                .SetLoops(numJumps * 2, LoopType.Yoyo)
                .OnStart(()=> startPosY = target.anchoredPosition.y);
            s.Append(DOTween.To(() => target.anchoredPosition, x => target.anchoredPosition = x, new Vector2(endValue.x, 0), duration)
                    .SetOptions(AxisConstraint.X, snapping).SetEase(Ease.Linear)
                ).Join(yTween)
                .SetTarget(target).SetEase(DOTween.defaultEaseType);
            s.OnUpdate(() => {
                if (!offsetYSet) {
                    offsetYSet = true;
                    offsetY = s.isRelative ? endValue.y : endValue.y - startPosY;
                }
                Vector2 pos = target.anchoredPosition;
                pos.y += DOVirtual.EasedValue(0, offsetY, s.ElapsedDirectionalPercentage(), Ease.OutQuad);
                target.anchoredPosition = pos;
            });
            return s;
        }

        public static class Utils
        {
            /// <summary>
            /// Converts the anchoredPosition of the first RectTransform to the second RectTransform,
            /// taking into consideration offset, anchors and pivot, and returns the new anchoredPosition
            /// </summary>
            public static Vector2 SwitchToRectTransform(RectTransform from, RectTransform to)
            {
                Vector2 localPoint;
                Vector2 fromPivotDerivedOffset = new Vector2(from.rect.width * 0.5f + from.rect.xMin, from.rect.height * 0.5f + from.rect.yMin);
                Vector2 screenP = RectTransformUtility.WorldToScreenPoint(null, from.position);
                screenP += fromPivotDerivedOffset;
                RectTransformUtility.ScreenPointToLocalPointInRectangle(to, screenP, null, out localPoint);
                Vector2 pivotDerivedOffset = new Vector2(to.rect.width * 0.5f + to.rect.xMin, to.rect.height * 0.5f + to.rect.yMin);
                return to.anchoredPosition + localPoint - pivotDerivedOffset;
            }
        }

        /// (NOTE 1: only uses the colors of the gradient, not the alphas - NOTE 2: creates a Sequence, not a Tweener).
        /// Also stores the image as the tween's target so it can be used for filtered operations</summary>
        /// <param name="gradient">The gradient to use</param><param name="duration">The duration of the tween</param>
        public static Sequence DOGradientColor(this Material target, Gradient gradient, float duration)
        {
            Sequence s = DOTween.Sequence();
            GradientColorKey[] colors = gradient.colorKeys;
            int len = colors.Length;
            for (int i = 0; i < len; ++i) {
                GradientColorKey c = colors[i];
                if (i == 0 && c.time <= 0) {
                    target.color = c.color;
                    continue;
                }
                float colorDuration = i == len - 1
                    ? duration - s.Duration(false) // Verifies that total duration is correct
                    : duration * (i == 0 ? c.time : c.time - colors[i - 1].time);
                s.Append(target.DOColor(c.color, colorDuration).SetEase(Ease.Linear));
            }
            s.SetTarget(target);
            return s;
        }

﻿using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
[RequireComponent(typeof(BoxCollider))]
public class CollapsingPlatform : MonoBehaviour
{
    [Header("Detection Settings (BoxCast)")]
    [SerializeField] private float castDistance = 2.0f;     // Độ cao khối hộp quét lên trời
    [SerializeField] private LayerMask detectionLayer;    // Layer nhận diện (xe)
    [SerializeField] private float boxSizeMultiplier = 0.95f; // Thu nhỏ nhẹ để tránh đụng bệ bên cạnh

    [Header("Timing Settings")]
    [SerializeField] private float totalTimeBeforeFall = 5.0f; // Tổng thời gian từ lúc chạm đến lúc rụng
    [SerializeField] private float shakeLeadTime = 2.0f;      // Thời gian sẽ rung trước khi rụng (Ví dụ: rung 2s cuối)
    [SerializeField] private float waitToHide = 3.0f;         // Chờ rụng xong mới ẩn
    [SerializeField] private float waitToRespawn = 3.0f;      // Chờ ẩn xong mới hiện lại

    [Header("Visual & Physics")]
    [SerializeField] private float shakeStrength = 0.15f;
    [SerializeField] private MeshRenderer visualRenderer;

    private Rigidbody rb;
    private BoxCollider boxCollider;
    private Vector3 originPos;
    private Quaternion originRot;
    private bool isProcessing = false;
    private Tweener shakeTweener;

    private void Awake()
    {
        rb = GetComponent<Rigidbody>();
        boxCollider = GetComponent<BoxCollider>();
        if (visualRenderer == null) visualRenderer = GetComponent<MeshRenderer>();

        originPos = transform.position;
        originRot = transform.rotation;

        rb.isKinematic = true;
    }

    private void Update()
    {
        if (isProcessing) return;

        // Bắn một khối hộp (BoxCast) có kích thước bằng với BoxCollider đang gắn trên bệ
        if (CheckBoxCast())
        {
            StartCoroutine(CollapseSequence());
        }
    }

    private bool CheckBoxCast()
    {
        // Lấy kích thước của Collider để làm kích thước cho BoxCast
        Vector3 halfExtents = new Vector3(
            (boxCollider.size.x * transform.lossyScale.x) / 2 * boxSizeMultiplier,
            0.1f, // Độ dày của khối quét
            (boxCollider.size.z * transform.lossyScale.z) / 2 * boxSizeMultiplier
        );

        // Bắn khối hộp lên phía trên
        return Physics.BoxCast(transform.position, halfExtents, Vector3.up, transform.rotation, castDistance, detectionLayer);
    }

    private IEnumerator CollapseSequence()
    {
        isProcessing = true;

        // 1. Giai đoạn im lặng ban đầu
        float silentDuration = totalTimeBeforeFall - shakeLeadTime;
        if (silentDuration > 0) yield return new WaitForSeconds(silentDuration);

        // 2. Giai đoạn rung lắc (Bắt đầu từ thời điểm bạn muốn cho đến khi rụng)
        // Dùng DOShakePosition và lưu vào biến để có thể dừng nếu cần
        shakeTweener = transform.DOShakePosition(shakeLeadTime, shakeStrength, 10, 90, false, true);
        yield return new WaitForSeconds(shakeLeadTime);

        // 3. Rụng
        rb.isKinematic = false;
        rb.useGravity = true;

        // 4. Chờ rụng xong rồi ẩn
        yield return new WaitForSeconds(waitToHide);
        visualRenderer.enabled = false;
        boxCollider.enabled = false;

        // 5. Chờ tái sinh
        yield return new WaitForSeconds(waitToRespawn);

        ResetPlatform();
    }

    private void ResetPlatform()
    {
        // Dừng mọi hiệu ứng DOTween đang chạy trên bệ này
        if (shakeTweener != null) shakeTweener.Kill();


        rb.isKinematic = true;
        rb.useGravity = false;

        transform.position = originPos;
        transform.rotation = originRot;

        if (VFXManager.Instance != null) VFXManager.Instance.PlaySpawnEffect(transform.position, 0.5f);

        visualRenderer.enabled = true;
        boxCollider.enabled = true;

        isProcessing = false;
    }

    // Vẽ khối hộp trong Editor để bạn căn chỉnh
    private void OnDrawGizmosSelected()
    {
        if (boxCollider == null) boxCollider = GetComponent<BoxCollider>();

        Gizmos.color = Color.yellow;
        Vector3 halfExtents = new Vector3(
            (boxCollider.size.x * transform.lossyScale.x) / 2 * boxSizeMultiplier,
            0.1f,
            (boxCollider.size.z * transform.lossyScale.z) / 2 * boxSizeMultiplier
        );

        Matrix4x4 rotationMatrix = Matrix4x4.TRS(transform.position, transform.rotation, transform.lossyScale);
        Gizmos.matrix = rotationMatrix;

        // Vẽ khối hộp đại diện cho BoxCast
        Gizmos.DrawWireCube(Vector3.up * castDistance, boxCollider.size * boxSizeMultiplier);
    }
}

    // [EN] This is called by VehicleSelectionManager
    public void StartGameWithSelections(List<VehicleDataSO> selections)
    {
        var globalManager = GlobalGameManager.Instance;

        for (int i = 0; i < selections.Count; i++)
        {
            if (i >= spawnPoints.Count) break;

            InputDevice device = globalManager.joinedPlayerDevices[i];
            GameObject selectedPrefab = selections[i].vehiclePrefab;

            // Spawn xe thật / Spawn real car
            PlayerInput newPlayer = PlayerInput.Instantiate(
                selectedPrefab,
                playerIndex: i,
                splitScreenIndex: -1,
                controlScheme: null,
                pairWithDevice: device
            );

            activePlayers.Add(newPlayer);
            playerLives.Add(newPlayer, startingLives);
            UpdateHealthUI(i, startingLives);

            RespawnPlayer(newPlayer, true);
        }
    }

    private void UpdateHealthUI(int playerIndex, int currentLives)
    {
        if (playerIndex < playerHealthTexts.Length && playerHealthTexts[playerIndex] != null)
            playerHealthTexts[playerIndex].text = $"{currentLives}";

    {
        if (!activePlayers.Contains(player)) return;

        if (VFXManager.Instance != null)
            VFXManager.Instance.PlayKillEffect(player.transform.position, 0.5f);

        int pIndex = player.playerIndex;
        playerLives[player]--;
        UpdateHealthUI(pIndex, playerLives[player]);

        if (playerLives[player] > 0)
        {
            RespawnPlayer(player, false);
        }
        else
        {
            activePlayers.Remove(player);
            player.gameObject.SetActive(false);


            CheckWinCondition();
        }
    }

    private void RespawnPlayer(PlayerInput player, bool isFirstSpawn)
    {
        CarVisual visual = player.GetComponent<CarVisual>();
        if (visual != null) visual.RepairCar();

        int id = player.playerIndex;
        var rb = player.GetComponent<Rigidbody>();

        if (rb != null && !isFirstSpawn)
        {
            rb.linearVelocity = Vector3.zero;
            rb.angularVelocity = Vector3.zero;
        }

        player.transform.position = spawnPoints[id].position;
        player.transform.rotation = spawnPoints[id].rotation;

        if (VFXManager.Instance != null)
            VFXManager.Instance.PlaySpawnEffect(spawnPoints[id].position, 0.5f);

        // [VI] 3. Đợi 0.3 giây, sau đó phóng to chiếc xe ra một cách "nhún nhảy" (OutBack)
        // [EN] 3. Wait 0.3 seconds, then scale the car up with a bouncy feel (OutBack)
        player.transform.DOScale(Vector3.one, 0.5f).SetDelay(0.3f).SetEase(Ease.OutBack);

        if (rb != null) rb.isKinematic = false;
    }

            float finalY = Mathf.Clamp(dis * 0.7f, minZoomIn, maxZoomOut);

        }

﻿using UnityEngine;
using TMPro;
using UnityEngine.UI;
using UnityEngine.InputSystem;

public class PlayerSelectionHandler : MonoBehaviour
{
    [Header("UI References")]
    public TextMeshProUGUI playerNameText;
    public TextMeshProUGUI vehicleNameText;
    public TextMeshProUGUI statsText;
    public Image vehicleImage;
    public GameObject readyOverlay; // Màn mờ hoặc chữ READY / Ready overlay or text

    [HideInInspector] public int playerIndex;
    [HideInInspector] public VehicleDataSO selectedVehicle;
    [HideInInspector] public bool isReady = false;

    [Header("Input Settings")]
    public float inputThreshold = 0.8f; // Ngưỡng nhận diện / Detection threshold
    private bool isAxisInUse = false;   // Biến khóa chống lặp / Anti-repeat lock variable

    private VehicleSelectionManager selectionManager;
    private int currentVehicleIndex = 0;

    public void Setup(int pIndex, VehicleSelectionManager manager)
    {
        playerIndex = pIndex;
        selectionManager = manager;
        playerNameText.text = $"P{playerIndex + 1}";
        readyOverlay.SetActive(false);

        UpdateUI();
    }

    // [VI] Nhận Input Trái/Phải để đổi xe / [EN] Receive Left/Right Input to change car
    public void OnNavigate(InputAction.CallbackContext context)
    {
        if (isReady) return; // Nếu đã Ready thì không cho đổi xe nữa

        Vector2 input = context.ReadValue<Vector2>();

        // [VI] Kiểm tra ngưỡng (ngưỡng 0.8 để tránh Joycon bị trôi)
        // [EN] Check threshold (0.8 threshold to avoid Joycon drift)
        if (Mathf.Abs(input.x) > inputThreshold)
        {
            // [VI] Chỉ thực hiện nếu trục CHƯA bị khóa
            // [EN] Only execute if the axis is NOT locked
            if (!isAxisInUse)
            {
                if (input.x > 0) ChangeVehicle(1);  // Gạt phải
                else if (input.x < 0) ChangeVehicle(-1); // Gạt trái

                isAxisInUse = true; // [VI] KHÓA: Đã đổi xe, phải nhả tay mới được đổi tiếp
            }
        }
        else
        {
            // [VI] MỞ KHÓA: Khi cần gạt thả về giữa (dưới ngưỡng 0.8)
            // [EN] UNLOCK: When stick returns to center (below 0.8 threshold)
            isAxisInUse = false;
        }
    }

    // [VI] Nhận Input nút (ví dụ: Phím A/South) để Ready / [EN] Receive button Input (e.g., A/South) to Ready
    public void OnSubmit(InputAction.CallbackContext context)
    {
        if (!context.performed) return;

        isReady = !isReady; // Toggle ready
        readyOverlay.SetActive(isReady);

        // Báo cho Manager kiểm tra / Tell Manager to check
        selectionManager.CheckAllReady();
    }

    private void ChangeVehicle(int direction)
    {
        var db = selectionManager.availableVehicles;
        currentVehicleIndex += direction;

        // Vòng lặp index / Loop index
        if (currentVehicleIndex < 0) currentVehicleIndex = db.Count - 1;
        if (currentVehicleIndex >= db.Count) currentVehicleIndex = 0;

        UpdateUI();
    }

    private void UpdateUI()
    {
        selectedVehicle = selectionManager.availableVehicles[currentVehicleIndex];
        vehicleNameText.text = selectedVehicle.vehicleName;
        statsText.text = $"Mass: {selectedVehicle.mass}\nSpeed: {selectedVehicle.speed}";

        if (selectedVehicle.vehicleIcon != null)
            vehicleImage.sprite = selectedVehicle.vehicleIcon;
    }
}

﻿using System.Collections;
using UnityEngine;

public class SpawnPad : MonoBehaviour
{
    [Header("Pad Settings")]
    [Tooltip("Thời gian chờ (giây) trước khi bệ ẩn đi / Time to wait (seconds) before hiding")]
    [SerializeField] private float hideDelay = 2.0f;

    [SerializeField] private MeshRenderer visualRenderer;

    private Coroutine hideCoroutine;
    private int playersOnPad = 0;

    private void Awake()
    {
        if (visualRenderer == null) visualRenderer = GetComponent<MeshRenderer>();
        if (visualRenderer != null) visualRenderer.enabled = false;

        Debug.Log($"[SpawnPad - {gameObject.name}] AWAKE: Bệ đệm đã khởi tạo và tàng hình.");
    }

    private void OnTriggerEnter(Collider other)
    {
        // Báo cáo ngay khi có bất kỳ vật gì chạm vào / Report immediately when anything touches
        Debug.Log($"[SpawnPad - {gameObject.name}] ENTER: Vật chạm vào là '{other.name}', Tag đang có: '{other.tag}'");

        // KIỂM TRA BẰNG TAG / CHECK BY TAG
        if (other.CompareTag("Player"))
        {
            playersOnPad++;
            Debug.Log($"[SpawnPad - {gameObject.name}] ++ Tăng số người: {playersOnPad}");
            ShowPad();
        }
    }

    private void OnTriggerStay(Collider other)
    {
        // Hàm bảo hiểm phòng trường hợp rơi quá nhanh
        // Insurance function in case of falling too fast
        if (playersOnPad <= 0 && other.CompareTag("Player"))
        {
            Debug.Log($"[SpawnPad - {gameObject.name}] STAY (BẢO HIỂM): Phát hiện Player '{other.name}' nhưng đếm = 0 -> Khôi phục!");
            playersOnPad = 1;
            ShowPad();
        }
    }

    private void OnTriggerExit(Collider other)
    {
        Debug.Log($"[SpawnPad - {gameObject.name}] EXIT: Vật rời đi là '{other.name}', Tag đang có: '{other.tag}'");

        if (other.CompareTag("Player"))
        {
            playersOnPad--;
            Debug.Log($"[SpawnPad - {gameObject.name}] -- Giảm số người: {playersOnPad}");

            if (playersOnPad <= 0)
            {
                playersOnPad = 0;
                if (hideCoroutine != null) StopCoroutine(hideCoroutine);

                Debug.Log($"[SpawnPad - {gameObject.name}] Bắt đầu đếm ngược {hideDelay}s để ẩn bệ.");
                hideCoroutine = StartCoroutine(HideRoutine());
            }
        }
    }

    private void ShowPad()
    {
        if (hideCoroutine != null)
        {
            StopCoroutine(hideCoroutine);
            hideCoroutine = null;
            Debug.Log($"[SpawnPad - {gameObject.name}] Đã hủy đếm ngược ẩn bệ.");
        }

        if (visualRenderer != null && !visualRenderer.enabled)
        {
            visualRenderer.enabled = true;
            Debug.Log($"[SpawnPad - {gameObject.name}] ĐÃ HIỆN BỆ!");
        }
    }

    private IEnumerator HideRoutine()
    {
        yield return new WaitForSeconds(hideDelay);

        if (visualRenderer != null)
        {
            visualRenderer.enabled = false;
            Debug.Log($"[SpawnPad - {gameObject.name}] ĐÃ ẨN BỆ sau {hideDelay}s!");
        }
    }
}

﻿using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;

public class VehicleSelectionManager : MonoBehaviour
{
    public static VehicleSelectionManager Instance;

    [Header("Database & UI")]
    public List<VehicleDataSO> availableVehicles;
    [SerializeField] private GameObject playerSelectionUIPrefab; // UI Panel Prefab
    [SerializeField] private Transform uiHorizontalLayoutGroup;  // Nơi chứa các UI Panel / Container for UI Panels

    [Header("Game Container")]
    [SerializeField] private GameObject selectionCanvas; // Tắt Canvas này khi vào game / Disable this when game starts

    private List<PlayerSelectionHandler> spawnedHandlers = new List<PlayerSelectionHandler>();

    private void Awake()
    {
        Instance = this;
    }

    private void Start()
    {
        SpawnUIPanels();
    }

    private void SpawnUIPanels()
    {
        var globalData = GlobalGameManager.Instance;
        if (globalData == null || globalData.joinedPlayerDevices.Count == 0) return;

        for (int i = 0; i < globalData.joinedPlayerDevices.Count; i++)
        {
            InputDevice device = globalData.joinedPlayerDevices[i];

            // [VI] Sinh ra UI Panel và gán device cho PlayerInput của UI đó
            // [EN] Spawn UI Panel and pair device to that UI's PlayerInput
            PlayerInput uiInput = PlayerInput.Instantiate(
                playerSelectionUIPrefab,
                playerIndex: i,
                controlScheme: null,
                splitScreenIndex: -1,
                pairWithDevice: device
            );

            uiInput.transform.SetParent(uiHorizontalLayoutGroup, false);

            PlayerSelectionHandler handler = uiInput.GetComponent<PlayerSelectionHandler>();
            handler.Setup(i, this);
            spawnedHandlers.Add(handler);
        }
    }

    public void CheckAllReady()
    {
        if (spawnedHandlers.Count == 0) return;

        foreach (var handler in spawnedHandlers)
        {
            if (!handler.isReady) return; // Chỉ 1 người chưa Ready là hủy / If even 1 is not ready, abort
        }

        Debug.Log("ALL PLAYERS READY! STARTING GAME...");
        StartGame();
    }

    private void StartGame()
    {
        // 1. Tắt màn hình chọn xe / Disable selection screen
        selectionCanvas.SetActive(false);

        // 2. Gom danh sách xe đã chọn / Collect selected vehicles
        List<VehicleDataSO> finalSelections = new List<VehicleDataSO>();
        foreach (var handler in spawnedHandlers)
        {
            finalSelections.Add(handler.selectedVehicle);

            // Xóa cái Input của UI để nhường Input cho Xe thật
            // Destroy the UI's Input to give Input control to the real Car
            Destroy(handler.gameObject);
        }

        // 3. Gọi GameManager_1 spawn xe / Tell GameManager_1 to spawn cars
        GameManager_1.Instance.StartGameWithSelections(finalSelections);
    }
}

﻿using Unity.Mathematics;
using UnityEditor.ShaderGraph.Internal;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.XR;
public enum RamTier { Tier_0, Tier_1, Tier_2, Tier_3 };
[RequireComponent(typeof(Rigidbody))]

public class CarPhysics : MonoBehaviour
{

    [Header("References")]
    [SerializeField] public Transform[] tireTransform;
    [SerializeField] public Transform[] flipTransform;
    [SerializeField] private Transform[] forcePoint;
    
    private Rigidbody rb;

    [Header("Suspension Setting")]
    [SerializeField] public float restDistance;
    [SerializeField] private float springStrength;
    [SerializeField] private float springDamping;

    [Header("Steering Setting")]
    [SerializeField] private AnimationCurve frontGripFactor;
    [SerializeField] private AnimationCurve rearGripFactor;
    [SerializeField] private float frontFrictionMutiply = 1;
    [SerializeField] private float rearFrictionMutiply = 1;
    [HideInInspector] public bool isDrift;
    [SerializeField] private float massCenter = 0.05f;


    [Header("Accelaration Setting")]
    [SerializeField] private float maxSteerAngle = 50f;
    [Tooltip("m/s")]
    [SerializeField] private float carMaxSpeed;
    [SerializeField] private LayerMask groundLayer;
    [SerializeField] private AnimationCurve powerCurve;
    [SerializeField] private AnimationCurve frictionCurve;
    [HideInInspector] public float carSpeed;


    [Header("Braking Setting")]
    [SerializeField] private AnimationCurve brakeCurve;


    [Header("Ram Setting")]
    [SerializeField] private bool isRamEnabled = true; 
    [SerializeField] private float minSpeedToRam = 5f; 
    [SerializeField] private float ramTierStep = 3f;
    [SerializeField] private float speedBoostPerTier = 2f;  
    [SerializeField] private float massShiftPerTier = -0.04f;
    private float count;
    public RamTier CurrentTier;
    private float baseMaxSpeed;
    private float baseMassCenter;


    [Header("Input Data")]
    private Vector2 moveDir;   
    private float brakeStre;
    private bool driftCall;

    void Start()
    {
       rb ??= GetComponent<Rigidbody>();
       rb.centerOfMass = new Vector3(0, massCenter, 0);
        baseMaxSpeed = carMaxSpeed;
        baseMassCenter = massCenter;
        CurrentTier = RamTier.Tier_0;
    }
    private void Update()
    {
        ApplyTurn();
        Debug.Log(CurrentTier);
    }
    private void FixedUpdate()
    {
        ApplyDriftForce();
        ApplyForce();
        ApplyFlipForce();
        ApplyRamSystem();
    }
    private void ApplyForce()
    {
        //Applying force for front tires
        for (int i = 0; i < tireTransform.Length / 2; i++)
        {
            RaycastHit hit;
            bool rayDidHit = Physics.Raycast(tireTransform[i].position,-tireTransform[i].up, out hit, restDistance, groundLayer);
            if (rayDidHit)
            {
                ApplySuspensionForce(tireTransform[i], hit);
                ApplySteeringForce(tireTransform[i], hit, i);
                ApplyAccelarationForce(tireTransform[i], hit);
                ApplyFrictionForce(tireTransform[i], hit);
                ApplyBrakingForce(tireTransform[i], hit);
            }
        }
        //Applying force for rear tires
        for (int i = tireTransform.Length - 1; i >= tireTransform.Length / 2; i--)
        {
            RaycastHit hit;
            bool rayDidHit = Physics.Raycast(tireTransform[i].position,-tireTransform[i].up, out hit, restDistance, groundLayer);
            if (rayDidHit)
            {
                ApplySuspensionForce(tireTransform[i], hit);
                ApplySteeringForce(tireTransform[i], hit, i);
                ApplyAccelarationForce(tireTransform[i], hit);
                ApplyFrictionForce(tireTransform[i], hit);
                ApplyBrakingForce(tireTransform[i], hit);


            }
        }
    }    
    private void ApplyTurn()
    {
        float v = moveDir.x;
        float angel = v * maxSteerAngle;
        for (int i = 0; i < tireTransform.Length / 2; i++)
        {
            tireTransform[i].localRotation = Quaternion.Euler(0, angel, 0);
        }
    }    
    private void ApplySuspensionForce(Transform tire, RaycastHit hit)
    {
        Vector3 springDir = tire.up;

        float offset = restDistance - hit.distance;

        float Vel = Vector3.Dot(springDir, rb.GetPointVelocity(tire.position));

        float suspensionForce = (offset * springStrength) - (Vel * springDamping);

        rb.AddForceAtPosition(springDir * suspensionForce, tire.position);
    }
    private void ApplySteeringForce(Transform tire, RaycastHit hit, int i)
    {
        //Xác định bánh trước hay sau
        bool isFront = i < 2;
        //Lấy hướng của lực
        Vector3 steeringDir = tire.right;

        //Lấy velocity bánh xe
        Vector3 tireWorldVelo = rb.GetPointVelocity(tire.position);

        //Tính tích vô hướng
        float steeringVel = Vector3.Dot(steeringDir, tireWorldVelo); // Xe đang trượt ngang bao nhiêu, âm hay dương

        //Sự thay đổi của velocity mà ta mong muốn
        //tireGrip chạy từ 0-1 là độ lớn, 0 là không grip, 1 là full grip
        float desireGripFactor = isFront ? ((frontGripFactor.Evaluate(Mathf.Abs(steeringVel))) * frontFrictionMutiply)
            : ((rearGripFactor.Evaluate(Mathf.Abs(steeringVel))) * rearFrictionMutiply);
        float desireVelChange = -steeringVel * desireGripFactor;

        //Gia tốc mong muốn
        float desireAccel = desireVelChange / Time.fixedDeltaTime;

        float tireMass = rb.mass / 4;

        rb.AddForceAtPosition((steeringDir * (tireMass * desireAccel)), tire.position);
 
    }    
    private void ApplyAccelarationForce(Transform tire, RaycastHit hit)
    {
        //Lấy input
        float h = moveDir.y;
        Vector3 accelDir = tire.forward;
        carSpeed = Vector3.Dot(accelDir, rb.linearVelocity);
        if (h == 0.0f) return;
        float normalizedSpeed = Mathf.Clamp01(Mathf.Abs(carSpeed) / carMaxSpeed);
        float availableTourqe = (carSpeed >= carMaxSpeed) ? 0 : powerCurve.Evaluate(normalizedSpeed) * h * 5000;
        //Debug.Log("Speed" + "" + carSpeed + "" + "m/s");

        rb.AddForceAtPosition(accelDir * availableTourqe, tire.position);
    }    
    private void ApplyFrictionForce(Transform tire, RaycastHit hit)
    {
        float h = moveDir.y;
        if (h != 0.0f) return;

        Vector3 frictionDir = tire.forward;

        Vector3 tireWorldVel = rb.GetPointVelocity(tire.position);

        float frictionVel = Vector3.Dot(frictionDir, tireWorldVel);

        float normalizedFrictionSpeed = Mathf.Clamp01(Mathf.Abs(frictionVel) / carMaxSpeed);

        float desireFrictionChange = -frictionVel * frictionCurve.Evaluate(normalizedFrictionSpeed);

        float desireFriction = desireFrictionChange / Time.fixedDeltaTime;
        //Debug.Log("Friction" + "" + desireFriction);
        float tireMass = rb.mass / 4;

        rb.AddForceAtPosition((frictionDir * (tireMass * desireFriction)) * frontFrictionMutiply, tire.position);
    }
    private void ApplyBrakingForce(Transform tire, RaycastHit hit)
    {
        float braking = brakeStre;
        if (braking == 0.0f) return;

        Vector3 frictionDir = tire.forward;

        Vector3 tireWorldVel = rb.GetPointVelocity(tire.position);

        float brakeVel = Vector3.Dot(frictionDir, tireWorldVel);

        float normalizedBrakeSpeedSpeed = Mathf.Clamp01(Mathf.Abs(brakeVel) / carMaxSpeed);

        float desireFrictionChange = -brakeVel * brakeCurve.Evaluate(normalizedBrakeSpeedSpeed);

        float desireFriction = desireFrictionChange / Time.fixedDeltaTime;

        float tireMass = rb.mass / 4;

        rb.AddForceAtPosition((frictionDir * (tireMass * desireFriction)), tire.position);
    }
    private void ApplyDriftForce()
    {
        if (driftCall)
        {
            frontFrictionMutiply = 1.4f;
            rearFrictionMutiply = 0.3f;
            isDrift = true;
        }
        else
        {
            frontFrictionMutiply = 1;
            rearFrictionMutiply = 1;
            isDrift = false;

        }

    }
    private void ApplyFlipForce()
    {
        for (int i = 1; i < flipTransform.Length; i++)
        {
            RaycastHit hit;
            bool rayDidHit = Physics.Raycast(flipTransform[i].position, flipTransform[i].forward, out hit, 1f, groundLayer);
            if(rayDidHit)
            {
                Debug.Log("Nằm");
                float offSet = 1 - hit.distance;

                float vel = Vector3.Dot(flipTransform[i].forward, rb.GetPointVelocity(flipTransform[i].position));

                float flipForce = (offSet * 20000f) - (vel * 800f);

                rb.AddForceAtPosition(-flipTransform[i].forward * flipForce, flipTransform[i].position);
            }
            bool rayDidUpHit = Physics.Raycast(flipTransform[0].position, flipTransform[0].up, out hit, 1f, groundLayer);
            if (rayDidUpHit)
            {
                Debug.Log("Nằm Ngửa");
                float offSet = 1 - hit.distance;
                float offSetUp = 1 - hit.distance;

                float vel = Vector3.Dot(flipTransform[i].forward, rb.GetPointVelocity(flipTransform[i].position));
                float velUp = Vector3.Dot(flipTransform[0].up, rb.GetPointVelocity(flipTransform[0].position));

                float flipForce = (offSet * 20000f) - (vel * 500f);

                float upFlipForce = (offSetUp * 5000f) - (velUp * 100f);

                rb.AddForceAtPosition(-flipTransform[0].up * upFlipForce, flipTransform[0].position);
                transform.Rotate(0,0,Mathf.Lerp(transform.rotation.z, 0, Time.deltaTime));
            }
        }

    }
    private void ApplyRamSystem()
    {
        if (!isRamEnabled)
        {
            count = 0;
            CurrentTier = RamTier.Tier_0;
            UpdatRamMutiply();
            return;
        }

        if (carSpeed >= minSpeedToRam) count += Time.deltaTime;
        else count = 0;

        float t1 = ramTierStep / 2f;       
        float t2 = t1 + ramTierStep;        
        float t3 = t2 + ramTierStep;


        if (count >= t3) CurrentTier = RamTier.Tier_3;
        else if (count >= t2) CurrentTier = RamTier.Tier_2;
        else if (count >= t1) CurrentTier = RamTier.Tier_1;
        else CurrentTier = RamTier.Tier_0;
        UpdatRamMutiply();
    }    
    private void UpdatRamMutiply()
    {
        int tierLevel = (int)CurrentTier;

        carMaxSpeed = baseMaxSpeed + (speedBoostPerTier * tierLevel);
        massCenter = baseMassCenter + (massShiftPerTier * tierLevel);

        rb.centerOfMass = new Vector3(0, massCenter, 0);
    }




    private void OnCollisionEnter(Collision col)
    {
        Vector3 hitPosi = col.GetContact(0).point;
        Vector3 localPosi = transform.InverseTransformPoint(hitPosi);
        Rigidbody rb = col.transform.GetComponent<Rigidbody>();

        if(isRamEnabled && localPosi.z > 1f)
        {
            if(rb)
            {
                RamEnemy(rb, col);
                if(CurrentTier >= RamTier.Tier_1) TriggerEnemyDamageVisual(rb);
                if(CurrentTier >= RamTier.Tier_3)
                {
                    VFXManager.Instance.PlayCollideEffect(hitPosi, 0.5f);

                }
                Debug.Log("Enemy");
            }
            else
            {
                RamConstruction(1f);
                if(CurrentTier >= RamTier.Tier_1) TriggerSelfDamageVisual();
                Debug.Log("Construct");

            }
        }    
    }
    private void RamConstruction(float mutiply)
    {
        float aganistForce = (rb.mass * carSpeed * 10f) * mutiply * ((int)CurrentTier + 0.2f);
        rb.AddForceAtPosition(transform.up * aganistForce * 0.6f, forcePoint[1].position);
        rb.AddForceAtPosition(-transform.forward * aganistForce * 1.2f, forcePoint[0].position);
    }    

    private void RamEnemy(Rigidbody rg, Collision col)
    {
        // [VI] 1. Tỷ lệ lợi thế khối lượng (Attacker / Defender)
        // Dùng Clamp để xe nhỏ nhất không bị vô dụng (0.4x) và xe to nhất không quá lố (2.5x)
        // [EN] 1. Mass advantage ratio (Attacker / Defender)
        // Clamp so the smallest car isn't useless (0.4x) and the biggest isn't broken (2.5x)
        float massAdvantage = Mathf.Clamp(rb.mass / rg.mass, 0.4f, 2.5f);

        // [VI] 2. Hệ số Tier: Ép khoảng cách từ 5m (Tier 1) lên 20m (Tier 3)
        // [EN] 2. Tier Multiplier: Force the distance from 5m (Tier 1) to 20m (Tier 3)
        float tierMultiplier = 1f;
        if (CurrentTier == RamTier.Tier_2) tierMultiplier = 2.5f;
        else if (CurrentTier == RamTier.Tier_3) tierMultiplier = 4.0f;

        // [VI] 3. Tính lực tổng: Động lượng (Mass * Speed) * Tier * Tỷ lệ khối lượng
        // [EN] 3. Total Force: Momentum (Mass * Speed) * Tier * Mass Ratio
        float baseMomentum = rb.mass * carSpeed * 0.5f; // Hằng số 0.5f để scale cho Impulse
        float ramForce = baseMomentum * tierMultiplier * massAdvantage;

        // [VI] 4. Hướng hất: Thêm một chút lực đẩy lên trên (y = 0.2f) để xe bay đẹp hơn
        // [EN] 4. Knockback direction: Add slight upward force (y = 0.2f) for better airtime
        Vector3 ramDir = (col.transform.position - transform.position).normalized;
        ramDir.y = 0.2f;

        // [VI] Sử dụng ForceMode.Impulse cho tác động tức thời (chuẩn vật lý va chạm)
        // [EN] Use ForceMode.Impulse for instant impact (accurate collision physics)
        rg.AddForce(ramDir * ramForce, ForceMode.Impulse);
        rg.AddTorque(UnityEngine.Random.insideUnitSphere * ramForce * 0.1f, ForceMode.Impulse);

        // [VI] 5. Phản lực cho xe tấn công (Recoil)
        // [EN] 5. Recoil for the attacker
        float recoilPenalty = Mathf.Clamp(rg.mass / rb.mass, 0.5f, 2f);
        RamConstruction(recoilPenalty);
    }

﻿using System.Collections.Generic;
using UnityEngine;
using DG.Tweening;

public class CarVisual : MonoBehaviour
{
    [Header("References")]
    [SerializeField] private Transform[] wheelMesh;
    [SerializeField] private ParticleSystem[] wheelSmokes;
    [SerializeField] private List<HingeJoint> damageableParts;

    [Header("Setting")]
    [SerializeField] private float wheelTourqeRatio = 0.35f;
    [SerializeField] private float wheelRadius = 0.3f;

    [Header("Evolution & Nitro Effects")]
    [SerializeField] private ParticleSystem nitroParticle; // Kéo Prefab Nitro vào đây / Drag Nitro Prefab here
    [SerializeField] private Color tier1Color = new Color(1f, 1f, 1f); // Trắng/Khói (White/Smoke)
    [SerializeField] private Color tier2Color = new Color(1f, 0f, 0f); // Đỏ (Red)
    [SerializeField] private Color tier3Color = new Color(0.6f, 0f, 1f); // Tím (Purple)

    private RamTier previousTier = RamTier.Tier_0;



    // Bộ nhớ bí mật để lưu setup gốc / Secret memory to store original setups
    private Dictionary<HingeJoint, JointLimits> originalLimits = new Dictionary<HingeJoint, JointLimits>();
    private List<HingeJoint> allOriginalParts = new List<HingeJoint>();
    private Dictionary<HingeJoint, Vector3> originalLocalPositions = new Dictionary<HingeJoint, Vector3>();
    private Dictionary<HingeJoint, Quaternion> originalLocalRotations = new Dictionary<HingeJoint, Quaternion>();



    private Rigidbody rb;
    private CarPhysics physicsScript;
    private float desireRotation;
    private void Awake()
    {
        physicsScript = GetComponent<CarPhysics>();
        rb = GetComponent<Rigidbody>();
        LockDamageableParts();
        if (nitroParticle != null)
            nitroParticle.Stop();

    }

    private void Update()
    {
        UpdateWheelMovement();
        ApplyDust();
        CheckTierEvolution();
    }


    private void UpdateWheelMovement()
    {
        Transform[] tireTrans = physicsScript.tireTransform;
        for (int i = 0; i < wheelMesh.Length; i++)
        {
            ApplyWheelLocation(i, tireTrans[i]);
            ApplyWheelRotation(i, tireTrans[i]);
        }
    }    
    private void ApplyWheelRotation(int i, Transform tire)
    {
        desireRotation = (physicsScript.carSpeed == 0.0f) ? 0 : (physicsScript.carSpeed * wheelTourqeRatio);
        wheelMesh[i].Rotate(desireRotation, 0, 0, Space.Self);
        
    }   
    private void ApplyWheelLocation(int i, Transform tire)
    {

            RaycastHit hit;
            bool rayDidHit = Physics.Raycast(tire.position, -tire.up, out hit, physicsScript.restDistance);
            if (rayDidHit)
            {
                wheelMesh[i].position = Vector3.Lerp(wheelMesh[i].position, tire.position + new Vector3(0, -(hit.distance - wheelRadius), 0), Time.deltaTime * 30f);
            }
    }   
    private void ApplyDust()
    {
        float carSpeed = physicsScript.carSpeed;

        foreach (var smoke in wheelSmokes)
        {
            var emission = smoke.emission;
            // Chỉ bật khói khi xe chạy nhanh hoặc khi bạn muốn (ví dụ đang phanh/drift)
            // Only enable smoke when speeding or braking/drifting
            emission.enabled = carSpeed > 5f || physicsScript.isDrift;
        }
    }

    // 1. Hàm khóa chặt lúc bắt đầu / Function to lock firmly at start
    public void LockDamageableParts()
    {
        if (damageableParts == null || damageableParts.Count == 0) return;

        foreach (HingeJoint part in damageableParts)
        {
            if (part != null)
            {
                part.useLimits = true;

                // LƯU LẠI setup gốc (Chỉ lưu 1 lần duy nhất lúc mới chơi) / SAVE original setup (Only save once at start)
                if (!originalLimits.ContainsKey(part))
                {
                    originalLimits[part] = part.limits;
                    allOriginalParts.Add(part); // Lưu vào danh sách gốc / Save to original list
                    originalLocalPositions[part] = part.transform.localPosition;
                    originalLocalRotations[part] = part.transform.localRotation;
                }

                // Ép góc về 0 để cửa dính chặt / Force angle to 0 so doors stick firmly
                JointLimits lockedLimits = part.limits;
                lockedLimits.min = 0f;
                lockedLimits.max = 0f;
                part.limits = lockedLimits;
            }
        }
    }

    // 2. Hàm xử lý khi bị tông (Dùng Clone Thế Thân) / Function to handle when rammed (Using Clone Debris)
    public void TakeVisualDamage()
    {
        if (damageableParts == null || damageableParts.Count == 0) return;

        int randomIndex = UnityEngine.Random.Range(0, damageableParts.Count);
        HingeJoint selectedPart = damageableParts[randomIndex];

        if (selectedPart != null)
        {
            if (selectedPart.limits.max == 0f && selectedPart.limits.min == 0f)
            {
                // Trạng thái 1: TRẢ LẠI giới hạn gốc / State 1: RESTORE original limits
                selectedPart.limits = originalLimits[selectedPart];
            }
            else
            {
                // Trạng thái 2: Đã lung lay rồi -> Rụng luôn / State 2: Already swinging -> Fall off completely

                // TẠO BẢN SAO RƠI XUỐNG ĐƯỜNG / CREATE CLONE TO FALL ON ROAD
                GameObject debris = Instantiate(selectedPart.gameObject, selectedPart.transform.position, selectedPart.transform.rotation);

                // FIX LỖI SIZE BỊ TO: Ép scale của bản sao bằng đúng scale tổng thể của cửa gốc
                // FIX ENLARGED SIZE BUG: Force clone's local scale to exactly match the global lossyScale of the original door
                debris.transform.localScale = selectedPart.transform.lossyScale;

                Destroy(debris.GetComponent<HingeJoint>()); // Đập vỡ bản lề của bản sao / Break clone's hinge

                // Giữ lại mảnh vỡ trên sàn đấu 20 giây cho chân thực / Keep debris on the arena for 20 seconds for realism
                Destroy(debris, 20f);

                // ẨN CỬA THẬT ĐI / HIDE THE REAL DOOR
                selectedPart.gameObject.SetActive(false);
                damageableParts.RemoveAt(randomIndex);
            }
        }
    }

    // 3. Hàm gọi khi xe hồi sinh / Function called when car respawns
    public void RepairCar()
    {
        damageableParts.Clear(); // Xóa danh sách cũ / Clear old list

        foreach (HingeJoint part in allOriginalParts)
        {
            if (part != null)
            {
                part.transform.localPosition = originalLocalPositions[part];
                part.transform.localRotation = originalLocalRotations[part];

                part.gameObject.SetActive(true);    // Hiện lại cánh cửa đã tàng hình / Unhide the invisible door
                part.limits = originalLimits[part]; // Trả lại setup gốc / Restore original setup
                damageableParts.Add(part);          // Đưa vào lại danh sách để tông tiếp / Re-add to active list
            }
        }

        LockDamageableParts(); // Khóa chặt lại y như mới! / Lock them tight like new!
    }



    private void CheckTierEvolution()
    {
        // Liên tục soi xem Tier bên Physics có thay đổi không / Continuously watch if Physics Tier changed
        if (physicsScript.CurrentTier != previousTier)
        {
            // Nếu Tier mới LỚN HƠN Tier cũ -> Tiến hóa!
            // If new Tier is GREATER THAN old Tier -> Evolve!
            if (physicsScript.CurrentTier > previousTier)
            {
                TriggerEvolutionPunch();
                UpdateNitroColor(physicsScript.CurrentTier);
            }
            // Nếu rớt đài về Tier 0 -> Tắt Nitro
            // If dropped back to Tier 0 -> Turn off Nitro
            else if (physicsScript.CurrentTier == RamTier.Tier_0)
            {
                if (nitroParticle != null && nitroParticle.isPlaying)
                    nitroParticle.Stop();
            }

            // Cập nhật lại bộ nhớ / Update memory
            previousTier = physicsScript.CurrentTier;

    private void followTarget()
    {
        if (target == null)
        {
            // Tự dọn dẹp bản thân nếu mục tiêu đã biến mất
            // Clean itself up if the target has disappeared
            Destroy(gameObject);
            return;
        }

        Quaternion lookRotate = Quaternion.LookRotation(dir);
        Vector3 roation = Quaternion.Lerp(transform.rotation, target.rotation, Time.deltaTime * 5f).eulerAngles;
        transform.rotation = Quaternion.Euler(0, roation.y, 0);

        float disThisFrame = speed * Time.deltaTime;
        if(dir.magnitude < disThisFrame)
        {
            HitTarget();
            return;
        }    
        transform.Translate(dir.normalized * speed * Time.deltaTime, Space.World);
    }

        //Debug.Log("Hit");

        }

    private void Start()
    {
        target = WayPoint.Points[0];
        // Thiết lập máu gốc và cập nhật thanh UI
        health = startHealth;
        if (healthBar != null)
        {
            healthBar.maxValue = startHealth;
            healthBar.value = health;
        }
    }

    private void Update()
    {
        Movement();
        if (healthBar != null)
        {
            // Ép thanh máu quay cùng hướng với góc nhìn của Camera
            // Force the health bar to match the Camera's rotation
            healthBar.transform.rotation = UnityEngine.Camera.main.transform.rotation;
        }
    }

    private void Movement()
    {
        Vector3 dir = target.position - transform.position;
        transform.Translate(dir.normalized * speed * Time.deltaTime, Space.World);
        if(Vector3.Distance(transform.position, target.position) <= 0.1f)
        {
            GetNextWaypoint();
        }
    }

    private void GetNextWaypoint()
    {

        if (wavepointIndex >= WayPoint.Points.Length - 1)
        {
            if (GameManager_2.instance != null)
            {
                GameManager_2.instance.TakeDamage(damageToPlayer);
            }

            Destroy(transform.gameObject);
            return;
        }
        wavepointIndex++;
        target = WayPoint.Points[wavepointIndex];
    }

    public void TakeDamage(int damge)
    {
        health -= damge;
        if (healthBar != null)
        {
            healthBar.value = health;
        }
        if (health <= 0)
        {
            Die();
        }
    }

    private void Die()
    {
        GameManager_2.instance.AddGold(goldReward);
        Destroy(gameObject);
    }

﻿using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using System.Linq; // BẮT BUỘC để dùng các lệnh Gom nhóm (GroupBy, OrderBy) / REQUIRED to use Grouping commands
using UnityEngine.UI;
public class EnemyCardHandler : MonoBehaviour
{
    public List<Card> stuckCards = new List<Card>();
    private Coroutine explodeTimer;
    [SerializeField] private float waitTime = 2f;

    [Header("Visual Feedback (UI & VFX)")]
    [SerializeField] private Image[] cardUI_Slots; // Mảng chứa 5 ô Image rỗng trên đầu quái / Array of 5 empty Image slots
    [SerializeField] private GameObject explosionVFX_Prefab; // Particle hiệu ứng nổ / Explosion Particle
    [SerializeField] private Transform cardLayoutParent;

    private void Start()
    {
        UpdateCardUI();
    }


    private void Update()
    {
        if (cardLayoutParent != null)
        {
            cardLayoutParent.rotation = UnityEngine.Camera.main.transform.rotation;
        }
    }
    public void AddCard(Card newCard)
    {
        stuckCards.Add(newCard);

        UpdateCardUI();

        if (explodeTimer != null) StopCoroutine(explodeTimer);

        if (stuckCards.Count >= 5)
        {
            Explode();
        }
        else
        {
            explodeTimer = StartCoroutine(CountdownToExplode());
        }
    }

    IEnumerator CountdownToExplode()
    {
        yield return new WaitForSeconds(waitTime);
        Explode();
    }

    private void Explode()
    {
        if (stuckCards.Count == 0) return;

        // 1. TÌM COMBO (Thám tử soi bài) 
        // 1. FIND COMBO (Detective inspects cards)
        PokerCombo finalCombo = EvaluatePokerCombo();

        // 2. IN LOG DEBUG (Tách riêng cho gọn)
        // 2. PRINT DEBUG LOG (Separated for cleanliness)
        PrintDebugLog(finalCombo);

        // 3. LẤY SÁT THƯƠNG TỪ QUAN TÒA (GameManager_2)
        // 3. GET DAMAGE FROM THE JUDGE (GameManager_2)
        int damage = GameManager_2.instance.GetComboDamage(finalCombo);


        TriggerExplosionVFX();
        // 4. TRỪ MÁU VÀ DỌN DẸP
        // 4. DEDUCT HEALTH AND CLEAN UP
        Enemy e = transform.GetComponent<Enemy>();
        if (e != null)
            e.TakeDamage(damage);

        stuckCards.Clear();
    }


    private void UpdateCardUI()
    {
        // Kiểm tra an toàn / Safety check
        if (cardUI_Slots == null || cardUI_Slots.Length == 0) return;

        for (int i = 0; i < cardUI_Slots.Length; i++)
        {
            if (i < stuckCards.Count)
            {
                // Lấy thông tin bài hiện tại / Get current card info
                Card currentCard = stuckCards[i];

                // Tự động ghép tên (VD: "Hearts_2", "Spades_14") / Auto-generate name
                string spriteName = $"{currentCard.rank}-{currentCard.suit}";

                // Tự động dò tìm ảnh trong thư mục Resources / Auto-load from Resources
                Sprite loadedSprite = Resources.Load<Sprite>($"Cards/{spriteName}");

                if (loadedSprite != null)
                {
                    cardUI_Slots[i].sprite = loadedSprite;
                    cardUI_Slots[i].gameObject.SetActive(true); // Bật ảnh lên / Enable image
                }
                else
                {
                    Debug.LogWarning($"Thiếu ảnh lá bài: {spriteName} trong thư mục Resources/Cards!");
                }
            }
            else
            {
                // Nếu chưa có bài ở vị trí này thì tắt ảnh đi / Disable if no card in this slot
                cardUI_Slots[i].gameObject.SetActive(false);
            }
        }
    }

    // =========================================================================
    // HÀM 2: KÍCH HOẠT HIỆU ỨNG VÀ XÓA BÀI
    // FUNCTION 2: TRIGGER VFX AND CLEAR CARDS
    // =========================================================================
    private void TriggerExplosionVFX()
    {
        // 1. Chạy Particle an toàn / Safe Particle Instantiation
        if (explosionVFX_Prefab != null)
        {
            // Sinh ra hiệu ứng cao hơn quái 1 chút / Spawn slightly above enemy
            GameObject vfx = Instantiate(explosionVFX_Prefab, transform.position + Vector3.up * 1f, Quaternion.identity);
            Destroy(vfx, 1.5f); // Tự dọn dẹp để không nặng máy / Auto cleanup
        }

        // 2. Tắt hết 5 ô Image bài đi / Disable all 5 card Images
        if (cardUI_Slots != null)
        {
            foreach (var img in cardUI_Slots)
            {
                if (img != null) img.gameObject.SetActive(false);
            }
        }
    }


    // =========================================================================
    // HÀM TÌM COMBO POKER (THE POKER BRAIN)
    // =========================================================================
    private PokerCombo EvaluatePokerCombo()
    {
        // GOM NHÓM BẰNG LINQ / GROUPING WITH LINQ:
        // GroupBy(c => c.rank): Gom các lá bài có cùng Số lại với nhau. / Group cards with the same Rank together.
        // OrderByDescending(g => g.Count()): Sắp xếp nhóm đông nhất lên đầu. / Sort so the largest group is first.
        var rankGroups = stuckCards.GroupBy(c => c.rank).OrderByDescending(g => g.Count()).ToList();

        // GroupBy(c => c.suit): Gom các lá bài cùng Chất (Cơ, Rô...) / Group cards by Suit (Hearts, Diamonds...)
        var suitGroups = stuckCards.GroupBy(c => c.suit).ToList();

        // KIỂM TRA ĐIỀU KIỆN ĐẶC BIỆT / CHECK SPECIAL CONDITIONS:
        // isFlush (Đồng chất): Phải có đủ 5 lá VÀ tất cả nằm chung 1 nhóm Chất.
        // isFlush: Must have exactly 5 cards AND all fall into 1 Suit group.
        bool isFlush = (stuckCards.Count == 5) && (suitGroups.Count == 1);
        bool isStraight = false;

        // isStraight (Sảnh): Phải có 5 lá, chia làm 5 nhóm Số khác nhau (không trùng).
        // isStraight: Must have exactly 5 cards, split into 5 different Rank groups (no duplicates).
        if (stuckCards.Count == 5 && rankGroups.Count == 5)
        {
            // Lấy ra các con số và xếp từ nhỏ đến lớn / Extract numbers and sort ascending (e.g., 4, 5, 6, 7, 8)
            var orderedRanks = stuckCards.Select(c => c.rank).OrderBy(r => r).ToList();

            // Nếu (Số lớn nhất - Số nhỏ nhất == 4) thì chắc chắn là chuỗi liên tiếp!
            // If (Highest rank - Lowest rank == 4) then it's definitely a consecutive sequence!
            if (orderedRanks.Last() - orderedRanks.First() == 4)
            {
                isStraight = true;
            }
        }

        // TIẾN HÀNH SO SÁNH TỪ MẠNH XUỐNG YẾU / BEGIN COMPARING FROM STRONGEST TO WEAKEST:

        // 1. Straight Flush: Vừa Sảnh vừa Đồng Chất / Both Straight and Flush
        if (isFlush && isStraight) return PokerCombo.StraightFlush;

        // 2. Four of a Kind: Nhóm đông nhất có 4 lá / Largest group has 4 cards
        if (rankGroups[0].Count() == 4) return PokerCombo.FourOfAKind;

        // 3. Full House: Nhóm 1 có 3 lá VÀ nhóm 2 có 2 lá / Group 1 has 3 cards AND Group 2 has 2 cards
        if (rankGroups.Count >= 2 && rankGroups[0].Count() == 3 && rankGroups[1].Count() == 2) return PokerCombo.FullHouse;

        // 4. Flush: 5 lá đồng chất / 5 cards same suit
        if (isFlush) return PokerCombo.Flush;

        // 5. Straight: 5 lá liên tiếp / 5 consecutive cards
        if (isStraight) return PokerCombo.Straight;

        // 6. Three of a Kind: Nhóm đông nhất có 3 lá / Largest group has 3 cards
        if (rankGroups[0].Count() == 3) return PokerCombo.ThreeOfAKind;

        // 7. Two Pair: Nhóm 1 có 2 lá VÀ nhóm 2 có 2 lá / Group 1 has 2 cards AND Group 2 has 2 cards
        if (rankGroups.Count >= 2 && rankGroups[0].Count() == 2 && rankGroups[1].Count() == 2) return PokerCombo.TwoPair;

        // 8. Pair: Nhóm đông nhất có 2 lá / Largest group has 2 cards
        if (rankGroups[0].Count() == 2) return PokerCombo.Pair;

        // 9. High Card: Rác (Mặc định) / Default garbage
        return PokerCombo.HighCard;
    }

    // =========================================================================
    // HÀM IN CONSOLE / CONSOLE PRINT FUNCTION
    // =========================================================================
    private void PrintDebugLog(PokerCombo combo)
    {
        string color = "white";
        switch (combo)
        {
            case PokerCombo.StraightFlush: color = "red"; break;
            case PokerCombo.FourOfAKind: color = "magenta"; break;
            case PokerCombo.FullHouse: color = "orange"; break;
            case PokerCombo.Flush: color = "cyan"; break;
            case PokerCombo.Straight: color = "green"; break;
            case PokerCombo.ThreeOfAKind: color = "yellow"; break;
            case PokerCombo.TwoPair: color = "lightblue"; break;
            case PokerCombo.Pair: color = "lightblue"; break;
        }

        Debug.Log($"<color={color}>Enemy Exploded with: {combo.ToString().ToUpper()}!</color>");
    }
}

        foreach (var item in comboDamageList)

                comboDamageDict.Add(item.comboType, item.damage);

        if (EventSystem.current.IsPointerOverGameObject())

        if (buildManager.GetTurretToBuild() == null) return;
        rend.material.color = hoverColor;
    }
    private void OnMouseExit()
    {
        rend.material.color = startColor;
    }

    private void ShowErrorMessage(string message)
    {
        if (errorTextPrefab != null)
        {
            // Sinh ra thông báo ngay trên đầu Node
            GameObject textObj = Instantiate(errorTextPrefab, transform.position + Vector3.up * 1.5f, UnityEngine.Camera.main.transform.rotation);

            var tmp = textObj.GetComponentInChildren<TMPro.TextMeshPro>();
            if (tmp != null) tmp.text = message;

            // Chạy hiệu ứng mượt mà
            StartCoroutine(AnimateErrorText(textObj));
        }
    }
    private IEnumerator AnimateErrorText(GameObject obj)
    {
        float duration = 1.8f;
        float elapsed = 0f;
        Vector3 startPos = obj.transform.position;
        Vector3 endPos = startPos + Vector3.up * 1.0f; // Bay lên cao thêm 1 mét

        var tmp = obj.GetComponentInChildren<TMPro.TextMeshPro>();

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            float percent = elapsed / duration;

            // Hiệu ứng bay lên (Smooth Step)
            obj.transform.position = Vector3.Lerp(startPos, endPos, percent);

            // Hiệu ứng mờ dần
            if (tmp != null)
            {
                Color c = tmp.color;
                c.a = Mathf.Lerp(1, 0, percent);
                tmp.color = c;
            }

            yield return null;
        }

        Destroy(obj); // Tự xóa để nhẹ máy
    }
}

﻿using UnityEngine;

public class Turret : MonoBehaviour
{
    [Header("Reference Setup")]
    [SerializeField]private Transform rotatePart;
    [SerializeField]private Transform firePoint;
    [SerializeField]private GameObject bulletPrefab;
    [Header("Shoot Settings")]
    [SerializeField]private float range;
    [SerializeField]private float fireRate;
    [SerializeField] private LayerMask enemyLayer;
    [HideInInspector]
    public Transform target;
    private float fireCountDown;
    [Header("Poker Setup")]
    // Lá bài mặc định khi vừa xây xong / Default card upon building
    public Card currentCard = new Card { rank = 2, suit = "Hearts" };

    private void Start()
    {
        InvokeRepeating("updateTarget", 0f, 0.5f);
    }
    private void Update()
    {
        updateVisual();
        updateShoot();
    }

    private void updateVisual()
    {
        if (target != null)
        {
            Vector3 dir = target.position - transform.position;
            Quaternion lookRotation = Quaternion.LookRotation(dir);
            Vector3 rotation = Quaternion.Lerp(rotatePart.rotation, lookRotation, Time.deltaTime * 5f).eulerAngles;
            rotatePart.rotation = Quaternion.Euler(0, rotation.y, 0);
        }    
    }    
    private void updateTarget()
    {
        Collider[] enemies = Physics.OverlapSphere(transform.position, range, enemyLayer);
        float shortestDis = Mathf.Infinity;
        GameObject nearestE = null;
        foreach (Collider e in enemies)
        {
            float disToE = Vector3.Distance(transform.position, e.transform.position);
            if(disToE < shortestDis)
            {
                shortestDis = disToE;
                nearestE = e.transform.gameObject;
            }    
        }    
        target = nearestE != null ? nearestE.transform : null;
    }
    private void updateShoot()
    {
        if (target == null)
            return;
        if (fireCountDown <= 0)
        {
            Shoot();
            fireCountDown = 1f / fireRate;
        }

        fireCountDown -= Time.deltaTime;
    }
    private void Shoot()
    {
        //Debug.Log("SHOOT");
        if (bulletPrefab == null) return;
        GameObject bulletGo = (GameObject)Instantiate(bulletPrefab, firePoint.position, rotatePart.rotation);
        Bullet bullet = bulletGo.GetComponent<Bullet>();

        if (bullet != null)
        {
            // Truyền luôn lá bài của Turret cho viên đạn!
            // Pass the Turret's card directly to the bullet!
            bullet.Seek(target, currentCard);
        }
    }

    public void SetCardRank(int newRank)
    {
        currentCard.rank = newRank;
        Debug.Log($"Turret Rank updated: {newRank}");
    }

    // Hàm đổi Chất / Function to change Suit
    public void SetCardSuit(string newSuit)
    {
        currentCard.suit = newSuit;
        Debug.Log($"Turret Suit updated: {newSuit}");
    }

    private void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.red;
        Gizmos.DrawWireSphere(transform.position, range);
    }

}

﻿using UnityEngine;
using System.Collections;
using TMPro;

public enum Difficulty { Easy, Medium, Hard, Terrible }

public class WaySpawner : MonoBehaviour
{
    [Header("Reference")]
    [SerializeField] private Transform enemyEasy;    // Prefab cơ bản / Base prefab
    [SerializeField] private Transform enemyMedium;  // Prefab trung bình / Medium prefab
    [SerializeField] private Transform enemyHard;    // Prefab khó / Hard prefab
    [SerializeField] private Transform spawnPoint;
    [SerializeField] private TextMeshProUGUI curWaveUI;
    [SerializeField] private TextMeshProUGUI maxWaveUI;

    [Header("Difficulty Settings")]
    public Difficulty currentDifficulty = Difficulty.Easy;
    [SerializeField] private int baseEnemyCount = 5;

    [Header("Wave Settings")]
    public int maxWave = 20;
    public int waveNumber = 0;

    private bool isWaveActive = false; // Check xem đang trong turn không / Check if wave is active

    private void Start()
    {
        if (maxWaveUI != null) maxWaveUI.text = maxWave.ToString();
        UpdateUI();
    }

    private void Update()
    {
        // Tự động kiểm tra nếu hết quái trên map thì kết thúc Turn
        // Automatically check if all enemies are gone to end the turn
        if (isWaveActive && GameObject.FindGameObjectsWithTag("Enemy").Length == 0)
        {
            isWaveActive = false;
            Debug.Log("Wave Clear! Ready for next turn.");
        }
    }



    // Hàm gọi từ nút bấm UI "Next Turn" / Function called from UI button
    public void NextTurn()
    {
        if (isWaveActive)
        {
            Debug.Log("Wave is still running! Cannot start next one.");
            return;
        }

        if (waveNumber >= maxWave)
        {
            Debug.Log("Final wave reached!");
            return;
        }

        StartCoroutine(SpawnWave());
    }



    IEnumerator SpawnWave()
    {
        isWaveActive = true;
        waveNumber++;
        UpdateUI();

        int count = CalculateEnemyCount();
        Transform prefabToUse = GetEnemyPrefab();

        for (int i = 0; i < count; i++)
        {
            SpawnEnemy(prefabToUse);
            yield return new WaitForSeconds(0.5f);
        }
    }



    private int CalculateEnemyCount()
    {
        int multiplier = 2; // Default Easy
        if (currentDifficulty == Difficulty.Medium) multiplier = 3;
        else if (currentDifficulty == Difficulty.Hard) multiplier = 5;
        else if (currentDifficulty == Difficulty.Terrible) multiplier = 8;

        return baseEnemyCount + (waveNumber * multiplier);
    }



    private Transform GetEnemyPrefab()
    {
        // Logic dò từ cao xuống thấp / Fallback logic from high to low
        if ((currentDifficulty == Difficulty.Hard || currentDifficulty == Difficulty.Terrible) && enemyHard != null)
            return enemyHard;

        if (currentDifficulty >= Difficulty.Medium && enemyMedium != null)
            return enemyMedium;

        return enemyEasy; // Mặc định trả về Easy nếu các cái khác null / Default to Easy
    }



    private void SpawnEnemy(Transform prefab)
    {
        if (prefab == null) return;
        Instantiate(prefab, spawnPoint.position, spawnPoint.rotation);
    }



    private void UpdateUI()
    {
        if (curWaveUI != null) curWaveUI.text = waveNumber.ToString();
    }
}

﻿using UnityEngine;
using UnityEngine.AI; // [VI] Thư viện bắt buộc cho NavMesh / [EN] Required library for NavMesh

[RequireComponent(typeof(NavMeshAgent))]
public class AIWander : MonoBehaviour
{
    [Header("Wander Settings")]
    [SerializeField] private float wanderRadius = 15f; // [VI] Bán kính tìm đường / [EN] Radius to find a path
    [SerializeField] private float minWaitTime = 1f;   // [VI] Thời gian đứng chờ tối thiểu / [EN] Minimum wait time
    [SerializeField] private float maxWaitTime = 4f;   // [VI] Thời gian đứng chờ tối đa / [EN] Maximum wait time

    [Header("Rotation Settings")]
    [SerializeField] private float rotationSpeed = 10f; // [VI] Tốc độ xoay mặt / [EN] Turn speed

    private NavMeshAgent agent;
    private float waitTimer;
    private bool isWaiting;

    private void Awake()
    {
        agent = GetComponent<NavMeshAgent>();

        // [VI] TẮT xoay tự động của NavMesh để chúng ta tự xoay cho mượt
        // [EN] DISABLE NavMesh auto-rotation so we can rotate it smoothly ourselves
        agent.updateRotation = false;
    }

    private void Start()
    {
        SetNewDestination();
    }

    private void Update()
    {
        HandleMovementAndWaiting();
        HandleSmoothRotation();
    }

    private void HandleMovementAndWaiting()
    {
        // [VI] Kiểm tra xem AI đã đến đích chưa (hoặc gần đến đích)
        // [EN] Check if the AI has reached its destination (or is very close)
        if (!agent.pathPending && agent.remainingDistance <= agent.stoppingDistance)
        {
            if (!isWaiting)
            {
                // [VI] Bắt đầu đứng chờ / [EN] Start waiting
                isWaiting = true;
                waitTimer = Random.Range(minWaitTime, maxWaitTime);
            }
            else
            {
                // [VI] Đếm ngược thời gian chờ / [EN] Countdown wait time
                waitTimer -= Time.deltaTime;
                if (waitTimer <= 0f)
                {
                    SetNewDestination();
                }
            }
        }
    }

    private void HandleSmoothRotation()
    {
        // [VI] Chỉ xoay khi AI đang thực sự di chuyển (vận tốc > 0)
        // [EN] Only rotate when the AI is actually moving (velocity > 0)
        if (agent.velocity.sqrMagnitude > Mathf.Epsilon)
        {
            // [VI] Lấy hướng đi hiện tại của Agent
            // [EN] Get the current movement direction of the Agent
            Vector3 moveDirection = agent.velocity.normalized;

            // [VI] Khóa trục Y để AI không bị ngửa lên trời khi đi dốc
            // [EN] Lock the Y axis so the AI doesn't tilt upward on slopes
            moveDirection.y = 0;

            // [VI] Tính toán và xoay mượt mà (Slerp)
            // [EN] Calculate and rotate smoothly (Slerp)
            Quaternion targetRotation = Quaternion.LookRotation(moveDirection);
            transform.rotation = Quaternion.Slerp(transform.rotation, targetRotation, rotationSpeed * Time.deltaTime);
        }
    }

    private void SetNewDestination()
    {
        // [VI] Lấy 1 điểm Random và lệnh cho Agent đi tới đó
        // [EN] Get a Random point and command the Agent to go there
        Vector3 randomPoint = GetRandomNavMeshPoint(transform.position, wanderRadius);
        agent.SetDestination(randomPoint);
        isWaiting = false;
    }

    private Vector3 GetRandomNavMeshPoint(Vector3 center, float radius)
    {
        // [VI] Lấy 1 tọa độ ngẫu nhiên trong khối cầu / [EN] Get a random coordinate within a sphere
        Vector3 randomDirection = Random.insideUnitSphere * radius;
        randomDirection += center;

        NavMeshHit hit;
        // [VI] Hàm cực kỳ quan trọng: Tìm điểm gần nhất nằm TRÊN NavMesh (Surface)
        // [EN] Crucial function: Finds the closest point ON the NavMesh (Surface)
        if (NavMesh.SamplePosition(randomDirection, out hit, radius, NavMesh.AllAreas))
        {
            return hit.position;
        }

        // [VI] Nếu lỗi (tìm không ra), đứng yên tại chỗ / [EN] If error (can't find), stand still
        return center;
    }
}

﻿using UnityEngine;
[RequireComponent(typeof(Rigidbody))]
public class CharacterController : MonoBehaviour
{
    [Header("Preference")]
    public UnityEngine.Camera cam;
    [Header("Setting Speed")]
    [SerializeField] private float maxSpeed;
    [SerializeField] private float accelaration;
    [SerializeField] private float maxAccelaration;
    [Header("Setting Camera")]
    [SerializeField] private float sensitive;
    [Header("Setting Rotation")]
    [SerializeField] private float rotationSpeed = 10f;
    [Header("Setting Gravity")]
    [SerializeField] private float maxDistanceCheckGravity;
    [SerializeField] private float mutiply;
    [SerializeField] private LayerMask groundLayer;
    [Header("Setting Attack")]
    [SerializeField] private float attackCooldown = 0.5f; // [VI] Thời gian hồi chiêu / [EN] Cooldown time
    [SerializeField] private GameObject bulletPrefab;     // [VI] Prefab đạn cho Seeker / [EN] Seeker bullet prefab
    [SerializeField] private float bulletSpeed = 20f;     // [VI] Vận tốc đạn / [EN] Bullet speed
    [SerializeField] private float seekerRange = 20f;
    [SerializeField] private float hiderAttackRadius = 1.5f;
    [SerializeField] private float hiderAttackOffset = 1.0f;
    private float lastAttackTime; // [VI] Lưu mốc thời gian đánh cuối / [EN] Store last attack time
    private float xRotation = 0f;
    private float yRotation = 0f;

    private Vector2 moveIn;
    private Vector2 lookIn;
    private Vector3 m_goalVel;

    private bool isGround;
    public bool isSeeker;
    private bool isStunned = false;
    private Rigidbody rb;
    private void Awake()
    {
        rb = GetComponent<Rigidbody>();
        rb.interpolation = RigidbodyInterpolation.Interpolate;
    }
    private void FixedUpdate()
    {
        groundCheck();
        moveControl();
        gravityControl();
    }
    private void Start()
    {
        Cursor.lockState = CursorLockMode.Locked;
        //cam ??= transform.GetComponentInChildren<Camera>();
    }
    void Update()
    {
        if(isSeeker)
        {
            cameraControl();
        }
        if (isSeeker && !isStunned)
        {
            cameraControl();
        }
    }
    private void moveControl()
    {
        if (isStunned)
        {
            rb.linearVelocity = new Vector3(0, rb.linearVelocity.y, 0);
            return;
        }

        Vector3 moveDiretion = new Vector3(moveIn.x, 0, moveIn.y);
        if (moveDiretion.magnitude > 1.0f)
            moveDiretion.Normalize();

        Quaternion camRotY = Quaternion.Euler(0, cam.transform.eulerAngles.y, 0);

        Vector3 moveDir = camRotY * moveDiretion;

        Vector3 goalVel = moveDir * maxSpeed;

        m_goalVel = Vector3.MoveTowards(m_goalVel, goalVel, accelaration * Time.fixedDeltaTime);

        Vector3 needAccel = (m_goalVel - rb.linearVelocity) / Time.fixedDeltaTime;

        needAccel = Vector3.ClampMagnitude(needAccel, maxAccelaration);

        rb.AddForce(Vector3.Scale(needAccel * rb.mass, new Vector3(1, 0, 1)));

        if (moveDir.magnitude > 0.1f && !isSeeker)
        {
            Quaternion targetRotation = Quaternion.LookRotation(moveDir);
            transform.rotation = Quaternion.Slerp(transform.rotation, targetRotation, rotationSpeed * Time.deltaTime);
        }
    }
    private void gravityControl()
    {
        float gravityForce = isGround ? 0 : (rb.mass * 10) * mutiply;
        rb.AddForce(-transform.up * gravityForce);
    }

    private void groundCheck()
    {
        Ray ray = new Ray(transform.position, -transform.up);
        if (Physics.Raycast(ray, out RaycastHit hit, maxDistanceCheckGravity, groundLayer, QueryTriggerInteraction.Ignore))
        {
            Debug.Log("isGround");
            isGround = true;
        }
        else
        {
            Debug.Log("noneGround");
            isGround = false;
        }
    }
    private void cameraControl()
    {
        cam.transform.position = transform.position + new Vector3(0, 1, 0);
        float hor = lookIn.x;
        float ver = lookIn.y;
        xRotation -= lookIn.y * sensitive * Time.deltaTime;
        yRotation += lookIn.x * sensitive * Time.deltaTime;
        xRotation = Mathf.Clamp(xRotation, -60f, 60f);
        cam.transform.localRotation = Quaternion.Euler(xRotation, yRotation, 0f);
        transform.localRotation = Quaternion.Euler(0f, yRotation,0f);
    }

    public void lookInput(Vector2 lookVal)
    {
        lookIn = lookVal;
    }

    public void moveInput(Vector2 moveVal)
    {
        moveIn = moveVal;
    }
    public void attackInput()
    {
        if (Time.time < lastAttackTime + attackCooldown) return;

        if (isSeeker)
        {
            if (cam == null || bulletPrefab == null) return;

            // [VI] 1. Tạo đạn tại vị trí Camera / [EN] 1. Instantiate bullet at Camera pos
            GameObject bullet = Instantiate(bulletPrefab, cam.transform.position + cam.transform.forward, cam.transform.rotation);

            // [VI] 2. Gán vận tốc cho đạn / [EN] 2. Set bullet velocity
            Rigidbody bulletRb = bullet.GetComponent<Rigidbody>();
            if (bulletRb != null)
            {
                bulletRb.linearVelocity = cam.transform.forward * bulletSpeed;
            }

            // [VI] 3. Tự hủy khi hết Range: Time = Distance / Speed
            // [EN] 3. Self-destruct at max range: Time = Distance / Speed
            float lifeTime = seekerRange / bulletSpeed;
            Destroy(bullet, lifeTime);

            lastAttackTime = Time.time;
        }
        else
        {
            // [VI] Logic cận chiến của Hider / [EN] Hider melee logic
            Vector3 attackPos = transform.position + transform.forward * hiderAttackOffset;
            Collider[] hitColliders = Physics.OverlapSphere(attackPos, hiderAttackRadius);
            foreach (var hitCollider in hitColliders)
            {
                CharacterController target = hitCollider.GetComponent<CharacterController>();
                if (target != null && target.isSeeker)
                {
                    GameManager_2_2.Instance.OnSeekerHit(target.gameObject);
                    lastAttackTime = Time.time;
                    break;
                }
            }
        }
    }
    public void ApplyStun(float duration)
    {
        StartCoroutine(StunRoutine(duration));
    }
    private System.Collections.IEnumerator StunRoutine(float duration)
    {
        isStunned = true;
        m_goalVel = Vector3.zero; // [VI] Dừng vận tốc mục tiêu / [EN] Stop goal velocity

        Debug.Log("<color=yellow>[STUN] Seeker bị đứng hình!</color>");

        yield return new WaitForSeconds(duration);

        isStunned = false;
        Debug.Log("<color=green>[STUN] Seeker đã hồi phục!</color>");
    }

    // [VI] Vẽ vòng tròn để bạn dễ căn chỉnh trong Scene khi chọn Hider
    // [EN] Draw a gizmo so you can easily adjust in Scene when Hider is selected
    private void OnDrawGizmosSelected()
    {
        if (!isSeeker)
        {
            Gizmos.color = Color.red;
            Gizmos.DrawWireSphere(transform.position + transform.forward * hiderAttackOffset, hiderAttackRadius);
        }

    private InputAction moveAction;
    private Vector2 moveValue;

    private InputAction lookAction;
    private Vector2 lookValue;

    private InputAction attackAction;

    private void Start()
    {
        Cursor.lockState = CursorLockMode.Locked;
        player ??= GetComponent<CharacterController>();
    }
    private void Awake()
    {
        inputActions = GetComponent<PlayerInput>();

        var map = inputActions.actions.FindActionMap("Player");

        moveAction = map.FindAction("Move", true);
        lookAction = map.FindAction("Look", true);
        attackAction = map.FindAction("Attack", true);
        attackAction.performed += ctx => GetAttackInput();
    }

    private void OnEnable()
    {
        inputActions.actions.FindActionMap("Player").Enable();
    }

    private void OnDisable()
    {
        inputActions.actions.FindActionMap("Player").Disable();
    }

    private void OnDestroy()
    {
        // [VI] Hủy đăng ký an toàn / [EN] Safe unregistration
        if (attackAction != null)
        {
            attackAction.performed -= ctx => GetAttackInput();
        }
    }

    void Update()
    {
        if (!canInput) return;

        GetMoveInput();
        GetLookInput();
    }

    private void GetMoveInput()
    {
        moveValue = moveAction.ReadValue<Vector2>();
        player.moveInput(moveValue);

    }

    private void GetLookInput()
    {

        if(player.isSeeker)
        {
            lookValue = lookAction.ReadValue<Vector2>();
            player.lookInput(lookValue);
        }    

    }
    private void GetAttackInput()
    {
        if (!canInput) return;

        // [VI] Cả Seeker và Hider đều có thể gọi hàm này / [EN] Both Seeker and Hider can call this
        if (player != null)
        {
            player.attackInput();
        }
    }

﻿using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;
using TMPro;

public class GameManager_2_2 : MonoBehaviour
{
    // [VI] Singleton để các Script khác dễ dàng gọi đến
    // [EN] Singleton for easy access from other scripts
    public static GameManager_2_2 Instance { get; private set; }

    [Header("Player Prefabs")]

    [SerializeField] private GameObject hiderPrefab;

    [SerializeField] private UnityEngine.Camera seekerCameraPrefab;

    [SerializeField] private List<Transform> seekerSpawnPoints; // [VI] List vị trí spawn của Seeker / [EN] Seeker spawn points
    [SerializeField] private List<Transform> hiderSpawnPoints;  // [VI] List vị trí spawn của Hider / [EN] Hider spawn points
    [SerializeField] private Collider baseArea;                 // [VI] BaseArea chỉ dùng cho Item và AI / [EN] BaseArea only for AI & Item

    [SerializeField] private float spawnHeightOffset = 1.5f;

    [Header("Game Settings")]
    [SerializeField] private float gameTime = 120f;

    [SerializeField] private GameObject itemPrefab;
    [SerializeField] private GameObject smokeParticlePrefab;
    [SerializeField] private float itemSpawnInterval = 15f;
    private float itemTimer;
    private Transform activeSeekerTransform;

    [SerializeField] private GameObject aiPrefab;
    [SerializeField] private int minAiCount = 5;
    [SerializeField] private int maxAiCount = 15;

    private float currentTime;

        else Destroy(gameObject);
    }

    private void Start()
    {
        SetupCameras();
        SpawnPlayers();
        SpawnAI();

        seekerLives = seekerMaxLives;
        currentTime = gameTime;
        isGameActive = true;
        itemTimer = itemSpawnInterval;
        UpdateHiderCountUI();
    }

    private void Update()
    {
        if (!isGameActive) return;

        currentTime -= Time.deltaTime;
        UpdateTimerUI();
        itemTimer -= Time.deltaTime;
        if (itemTimer <= 0f)
        {
            SpawnItem();
            itemTimer = itemSpawnInterval;
        }

        if (currentTime <= 0)
        {
            currentTime = 0;
            EndGame(false);
        }
    }

    private void UpdateTimerUI()
    {
        if (timerText != null)
        {
            int minutes = Mathf.FloorToInt(currentTime / 60);
            int seconds = Mathf.FloorToInt(currentTime % 60);
            timerText.text = string.Format("{0:00}:{1:00}", minutes, seconds);
        }
    }

    private void UpdateHiderCountUI()
    {
        if (hiderCountText != null)
        {
            hiderCountText.text = $"Hiders: {aliveHidersCount}";
        }
    }

    private void SetupCameras()
    {
        if (seekerCameraPrefab != null)
        {
            activeSeekerCam = Instantiate(seekerCameraPrefab);
            activeSeekerCam.rect = new Rect(0f, 0f, 0.5f, 1f);
        }
        if (hiderCameraPrefab != null)
        {
            activeHiderCam = Instantiate(hiderCameraPrefab);
            activeHiderCam.rect = new Rect(0.5f, 0f, 0.5f, 1f);

            // [VI] Truyền Sàn nhà (baseArea) vào Camera / [EN] Pass the Floor (baseArea) into the Camera
            CameraOrbit orbitScript = activeHiderCam.GetComponent<CameraOrbit>();
            if (orbitScript != null && baseArea != null)
            {
                orbitScript.SetStaticTarget(baseArea.transform);
            }
        }
    }


    private void SpawnPlayers()
    {
        List<InputDevice> devices = GlobalGameManager.Instance.joinedPlayerDevices;
        if (devices == null || devices.Count == 0)
        {
            Debug.LogWarning("[GAME] Không có thiết bị nào kết nối! / No devices connected!");
            return;
        }

        int seekerIndex = Random.Range(0, devices.Count);
        aliveHidersCount = devices.Count - 1;

        // [VI] Tạo list tạm để bốc điểm Hider không bị trùng
        // [EN] Create a temp list to pick Hider points without overlap
        List<Transform> availableHiderSpawns = new List<Transform>(hiderSpawnPoints);

        for (int i = 0; i < devices.Count; i++)
        {
            InputDevice device = devices[i];

            if (i == seekerIndex)
            {
                PlayerInput seekerInput = PlayerInput.Instantiate(seekerPrefab, pairWithDevice: device);
                activeSeekerTransform = seekerInput.transform;

                Transform chosenSeekerSpawn = null;
                if (seekerSpawnPoints != null && seekerSpawnPoints.Count > 0)
                {
                    chosenSeekerSpawn = seekerSpawnPoints[Random.Range(0, seekerSpawnPoints.Count)];
                }

                ConfigurePlayer(seekerInput, true, chosenSeekerSpawn);
            }
            else
            {
                PlayerInput hiderInput = PlayerInput.Instantiate(hiderPrefab, pairWithDevice: device);

                Transform chosenHiderSpawn = null;
                if (availableHiderSpawns.Count > 0)
                {
                    int randIndex = Random.Range(0, availableHiderSpawns.Count);
                    chosenHiderSpawn = availableHiderSpawns[randIndex];
                    availableHiderSpawns.RemoveAt(randIndex);
                }

                ConfigurePlayer(hiderInput, false, chosenHiderSpawn);
            }
        }
    }

    private void ConfigurePlayer(PlayerInput pInput, bool isSeeker, Transform assignedSpawn)
    {
        CharacterController cc = pInput.GetComponent<CharacterController>();
        Rigidbody rb = pInput.GetComponent<Rigidbody>();

        if (cc == null) return;

        Vector3 finalPosition = pInput.transform.position;
        Quaternion finalRotation = pInput.transform.rotation;

        if (assignedSpawn != null)
        {
            finalPosition = assignedSpawn.position;
            finalRotation = assignedSpawn.rotation;
        }

        if (isSeeker)
        {
            cc.cam = activeSeekerCam;
            cc.isSeeker = true;
        }
        else
        {
            cc.cam = activeHiderCam;
            cc.isSeeker = false;

        }

        // [VI] ÉP VỊ TRÍ VẬT LÝ / [EN] FORCE PHYSICS POSITION
        if (rb != null)
        {
            rb.position = finalPosition;
            rb.rotation = finalRotation;
        }
        else
        {
            pInput.transform.position = finalPosition;
            pInput.transform.rotation = finalRotation;
        }
    }

    private void SpawnAI()
    {
        if (aiPrefab == null || baseArea == null) return;
        int aiToSpawn = Random.Range(minAiCount, maxAiCount + 1);

        for (int i = 0; i < aiToSpawn; i++)
        {
            Vector3 spawnPos = GetRandomPositionInBase();
            Instantiate(aiPrefab, spawnPos, Quaternion.identity);
        }
    }

    private void SpawnItem()
    {
        if (itemPrefab == null) return;
        Vector3 spawnPos = GetRandomPositionInBase();
        Instantiate(itemPrefab, spawnPos, Quaternion.identity);
    }

    // [VI] Hàm này giờ CHỈ DÙNG cho AI và Item / [EN] This function is now ONLY USED for AI and Items
    private Vector3 GetRandomPositionInBase()
    {
        if (baseArea == null) return Vector3.zero;
        Bounds bounds = baseArea.bounds;
        float rangeX = Mathf.Max(0, bounds.extents.x - spawnMargin);
        float rangeZ = Mathf.Max(0, bounds.extents.z - spawnMargin);
        float randomX = bounds.center.x + Random.Range(-rangeX, rangeX);
        float randomZ = bounds.center.z + Random.Range(-rangeZ, rangeZ);
        return new Vector3(randomX, bounds.max.y + spawnHeightOffset, randomZ);
    }

    public void OnHiderCaught(GameObject hiderObj)
    {
        if (!isGameActive) return;
        if (VFXManager.Instance != null)
        {
            VFXManager.Instance.PlaySpawnEffect(hiderObj.transform.position, 0.5f);
        }

        Debug.Log($"[GAME] Hider {hiderObj.name} đã bị bắt! / caught!");
        Destroy(hiderObj);
        aliveHidersCount--;
        UpdateHiderCountUI();

        if (aliveHidersCount <= 0)
        {
            EndGame(true);
        }
    }

    public void OnSeekerHit(GameObject seekerObj)
    {
        if (!isGameActive) return;

        seekerLives--;
        Debug.Log($"<color=orange>[GAME] Seeker bị đánh! Mạng còn lại: {seekerLives}</color>");

        ActivateSmokeAtPosition(seekerObj.transform.position, 3f);

        CharacterController seekerCC = seekerObj.GetComponent<CharacterController>();
        if (seekerCC != null)
        {
            seekerCC.ApplyStun(seekerStunDuration);
        }

        if (seekerLives <= 0)
        {
            Debug.Log("<color=red>[GAME] Seeker đã hết mạng!</color>");
            Destroy(seekerObj);
            EndGame(false);
        }
    }

    public void ActivateSmokeAtPosition(Vector3 position, float duration)
    {
        if (smokeParticlePrefab != null)
        {
            Vector3 spawnPos = position + new Vector3(0, 1f, 0);
            GameObject smoke = Instantiate(smokeParticlePrefab, spawnPos, Quaternion.identity);

            ParticleSystem ps = smoke.GetComponent<ParticleSystem>();
            if (ps != null)
            {
                var mainModule = ps.main;
                //mainModule.duration = duration;
                ps.Play();
            }
            Destroy(smoke, duration);
        }
    }

    public void ActivateSmokeOnSeeker(float times)
    {
        if (activeSeekerTransform != null)
        {
            ActivateSmokeAtPosition(activeSeekerTransform.position, times);
        }
    }

    private void EndGame(bool seekerWon)
    {
        isGameActive = false;
        if (seekerWon) Debug.Log("<color=red>SEEKER WIN!</color>");
        else Debug.Log("<color=blue>HIDER WIN!</color>");
    }
}

    private void Start()
    {
        count = maxTime;
        rb = GetComponent<Rigidbody>();
    }
    void Update()
    {
        countExplore();
        updateUI();
        ApplyFrictionForce();
    }
    void LateUpdate()
    {
        canvasUI.LookAt(canvasUI.position + camTransform.forward);
    }
    private void countExplore()
    {
        if (count > 0)
        {
            count -= Time.deltaTime;

        }
        else
        {
        }    
    }    
    private void ApplyFrictionForce()
    {
        Vector3 horizontalVelocity = new Vector3(rb.linearVelocity.x, 0, rb.linearVelocity.z);

        if (horizontalVelocity.magnitude > 0.5f)
        {
            float frictionCoefficient = 0.5f;
            Vector3 frictionForce = -horizontalVelocity * frictionCoefficient;

            rb.AddForce(frictionForce, ForceMode.Acceleration);
        }
    }

    private void updateUI()
    {
        canvasUI.position = Vector3.Lerp(canvasUI.position, transform.position + new Vector3(0, 5, 0), Time.deltaTime * 10f);
        int displayUi = Mathf.CeilToInt(count);
        countUI.text = displayUi.ToString();
    }    
    private void OnCollisionEnter(Collision other)
    {
        if (other.gameObject.CompareTag("Player"))
        {
            count = maxTime;
        }
    }
}

﻿using UnityEngine;

public class FrogCamera : MonoBehaviour
{
    [Header("Target Settings")]
    [SerializeField] private Transform target;        // Con ếch (Player)
    [SerializeField] private Vector3 targetOffset;    // Độ lệch tâm (ví dụ: nhìn cao hơn đầu ếch tí)

    [Header("Distance Settings")]
    [SerializeField] private float distance = 5f;     // Khoảng cách từ cam đến ếch

    [Header("Rotation Settings")]
    [SerializeField] private float mouseSensitivity = 0.2f;
    [SerializeField] private float smoothTime = 0.12f;    // Độ trễ mượt mà (Smoothing)
    [SerializeField] private Vector2 pitchLimits = new Vector2(-20f, 80f); // Giới hạn ngước lên/ngó xuống

    [Header("Collision Settings")]
    [SerializeField] private LayerMask collisionLayer; // Chọn Layer của tường và sàn
    [SerializeField] private float collisionRadius = 0.2f; // Độ dày của tia dò
    [SerializeField] private float collisionPadding = 0.1f; // Khoảng cách đệm để cam không sát tường quá
    [SerializeField] private float collisionSmoothTime = 10f;

    private Vector2 lookInput;
    private float yaw;   // Xoay quanh trục Y (trái/phải)
    private float pitch; // Xoay quanh trục X (lên/xuống)

    private Vector3 currentRotation;
    private Vector3 rotationSmoothVelocity;

    private Vector3 currentPosition;
    private Vector3 positionSmoothVelocity;

    private float currentDistance;

    private void Start()
    {
        // Khởi tạo góc quay hiện tại dựa trên Rotation của Camera trong Scene
        Vector3 angles = transform.eulerAngles;
        yaw = angles.y;
        pitch = angles.x;

        currentDistance = distance;
    }

    // Nhận dữ liệu từ FrogInput gửi sang
    public void SetLookInput(Vector2 input)
    {
        lookInput = input;
    }

    private void LateUpdate()
    {
        if (target == null) return;

        HandleRotation();
        HandlePosition();
    }

    private void HandleRotation()
    {
        // Cộng dồn input vào góc quay
        yaw += lookInput.x * mouseSensitivity;
        pitch -= lookInput.y * mouseSensitivity;

        // Giới hạn góc ngước lên/xuống để không bị lộn ngược camera
        pitch = Mathf.Clamp(pitch, pitchLimits.x, pitchLimits.y);

        // Làm mượt chuyển động xoay (Damping)
        Vector3 targetRot = new Vector3(pitch, yaw, 0);
        currentRotation = Vector3.SmoothDamp(currentRotation, targetRot, ref rotationSmoothVelocity, smoothTime);

        transform.eulerAngles = currentRotation;
    }

    private void HandlePosition()
    {
        // 1. Tính toán vị trí lý tưởng (Vị trí camera muốn đứng nếu không có tường)
        Vector3 targetPos = target.position + targetOffset;
        Vector3 direction = -transform.forward;
        Vector3 desiredPosition = targetPos + direction * distance;

        // 2. Dò tìm va chạm (SphereCast)
        RaycastHit hit;
        float adjustedDistance = distance;

        // Bắn một khối cầu từ nhân vật ra hướng Camera
        if (Physics.SphereCast(targetPos, collisionRadius, direction, out hit, distance, collisionLayer))
        {
            // Nếu chạm tường, khoảng cách mới = khoảng cách từ nhân vật đến điểm chạm - một chút đệm
            adjustedDistance = hit.distance - collisionPadding;
        }

        // 3. Làm mượt khoảng cách (để cam không bị giật cái rụp khi gặp vật cản)
        currentDistance = Mathf.Lerp(currentDistance, adjustedDistance, Time.deltaTime * collisionSmoothTime);

        // 4. Cập nhật vị trí cuối cùng
        transform.position = targetPos + direction * currentDistance;
    }
}

    public UnityEngine.Camera cam;
    private Rigidbody rb;

    [Header("Suspension (Float) Settings")]
    [SerializeField] private float rideHeight = 1.5f; // "offset" from center
    [SerializeField] private float raycastLength = 2.0f;
    [SerializeField] private float springStrength = 250f;
    [SerializeField] private float springDamper = 15f;
    [SerializeField] private LayerMask groundLayer;

    [Header("Movement Settings")]
    [SerializeField] private float maxSpeed = 8f;
    [SerializeField] private float accelaration = 60f;
    [SerializeField] private float maxAccelaration = 150f;

    [Header("Jump Settings")]
    [SerializeField] private float jumpForce = 15f;
    [SerializeField] private float jumpUpwardMultiplier = 1.5f; // [VI] MỚI: Trọng lực khi giữ nút / [EN] NEW: Gravity when holding
    [SerializeField] private float fallMultiplier = 4f; // The "down force" when falling
    [SerializeField] private float lowJumpMultiplier = 6f; // The "down force" when releasing jump early
    [SerializeField] private float coyoteTime = 0.15f;
    [SerializeField] private float jumpBufferTime = 0.15f;

    [Header("Kinematic Heading")]
    [SerializeField] private float turnSpeed = 30f; // Tốc độ xoay mặt (độ/giây) / Turn speed (degrees/sec)

    [Header("Physics Upright")]
    [SerializeField] private float uprightSpringStrength = 300f;
    [SerializeField] private float uprightSpringDamper = 30f;

    [Header("Health Settings")]
    public float maxHealth = 100f;
    private float currentHealth;

    // State Variables
    private Vector3 m_goalVel;
    private bool isGrounded;
    private float coyoteTimeCounter;
    private float jumpBufferCounter;
    private Vector2 currentMoveInput;
    private bool isJumpHeld;
    private float timeSinceJump = 1f;
    Quaternion targetRotation;

    public bool isGrappling = false; // Biến này sẽ được FrogTongue bật/tắt
    private void Awake()
    {
        rb = GetComponent<Rigidbody>();
        rb.useGravity = true;
        currentHealth = maxHealth;
    }
    private void Update()
    {
        timeSinceJump += Time.deltaTime; // Cứ đếm thời gian trôi qua
        HandleJumpTimers();
    }
    private void FixedUpdate()
    {
        // Nếu đang bị lưỡi kéo bay đi thì không tính toán các lực di chuyển/nhún khác
        if (isGrappling) return;

        HandleSuspension();
        HandleMovement();
        HandleJumpAction();
        HandleCustomGravity();
        //HandleUprightForce();
    }
    private void HandleSuspension()
    {
        Ray ray = new Ray(transform.position, Vector3.down);
        if (Physics.Raycast(ray, out RaycastHit hit, raycastLength, groundLayer))
        {
            if (timeSinceJump > 0.2f)
            {
                isGrounded = true;
                coyoteTimeCounter = coyoteTime;
            }

            // TẠM NGẮT lò xo không cho nó tính toán lực kéo xuống khi nhân vật đang lấy đà phóng lên
            if (timeSinceJump <= 0.2f) return;
            // Tính khoảng cách nén của "lò xo"
            // Khoảng cách thực tế so với khoảng cách mong muốn (rideHeight)
            float offset = rideHeight - hit.distance;

            // Vận tốc nén (vận tốc trục Y)
            float velocityY = rb.linearVelocity.y;

            // Tính toán lực nổi (Lực lò xo + Lực cản damper)
            float springForce = (offset * springStrength) - (velocityY * springDamper);

            // Áp dụng lực đẩy con ếch lên
            rb.AddForce(Vector3.up * springForce);

            // [PHẦN QUAN TRỌNG CHO GAME FEEL: DOWNWARD FORCE LÊN VẬT THỂ]
            // Áp dụng lực đè ngược lại lên vật thể bên dưới (nếu nó là lá sen/thùng gỗ có Rigidbody)
            if (hit.rigidbody != null)
            {
                hit.rigidbody.AddForceAtPosition(Vector3.down * springForce, hit.point);
            }
        }
        else
        {
            isGrounded = false;
        }
    }
    private void HandleMovement()
    {
        Vector2 moveIn = currentMoveInput;
        Vector3 moveDiretion = new Vector3(moveIn.x, 0, moveIn.y);

        if (moveDiretion.magnitude > 1.0f)
            moveDiretion.Normalize();

        Quaternion camRotY = Quaternion.Euler(0, cam.transform.eulerAngles.y, 0);
        Vector3 moveDir = camRotY * moveDiretion;
        Vector3 goalVel = moveDir * maxSpeed;

        m_goalVel = Vector3.MoveTowards(m_goalVel, goalVel, accelaration * Time.fixedDeltaTime);
        Vector3 needAccel = (m_goalVel - rb.linearVelocity) / Time.fixedDeltaTime;
        needAccel = Vector3.ClampMagnitude(needAccel, maxAccelaration);

        rb.AddForce(Vector3.Scale(needAccel * rb.mass, new Vector3(1, 0, 1)));

        if (moveDir.magnitude > 0.1f)
        {
            targetRotation = Quaternion.LookRotation(moveDir);
            transform.rotation = Quaternion.Slerp(transform.rotation, targetRotation,  turnSpeed * Time.deltaTime);
        }

    }

    private void HandleJumpTimers()
    {
        // Giảm dần bộ đếm Coyote Time nếu rời khỏi mặt đất
        if (!isGrounded) coyoteTimeCounter -= Time.deltaTime;
        jumpBufferCounter -= Time.deltaTime;
    }
    private void HandleJumpAction()
    {
        // Kiểm tra xem có lệnh nhảy trong Buffer và còn Coyote Time không
        if (jumpBufferCounter > 0f && coyoteTimeCounter > 0f)
        {
            // Reset vận tốc trục Y trước khi nhảy để nhảy luôn đều nhau
            rb.linearVelocity = new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);

            // Tiêu thụ bộ đếm để không nhảy liên tục
            jumpBufferCounter = 0f;
            coyoteTimeCounter = 0f;

            timeSinceJump = 0f;
        }
    }
    private void HandleCustomGravity()
    {
        if (isGrounded) return;

        if (rb.linearVelocity.y < 0)
        {
            // [VI] Đang rơi xuống / [EN] Falling down
            rb.AddForce(Vector3.down * (fallMultiplier - 1) * Physics.gravity.magnitude * rb.mass);
        }
        else if (rb.linearVelocity.y > 0)
        {
            if (!isJumpHeld)
            {
                // [VI] Nhả nút nhảy sớm (Cắt ngắn cú nhảy) / [EN] Released jump early (Cut jump short)
                rb.AddForce(Vector3.down * (lowJumpMultiplier - 1) * Physics.gravity.magnitude * rb.mass);
            }
            else
            {
                // [VI] Đang giữ nút nhảy (Kéo đầm lại để không bị Moon Jump)
                // [EN] Holding jump button (Pull down slightly to prevent Moon Jump)
                rb.AddForce(Vector3.down * (jumpUpwardMultiplier - 1) * Physics.gravity.magnitude * rb.mass);
            }
        }
    }
    public void SetMoveInput(Vector2 move)
    {
        currentMoveInput = move;
    }
    public void TriggerJumpDown()
    {
        // Khi FrogInput báo có người bấm nhảy, ta kích hoạt Jump Buffer
        jumpBufferCounter = jumpBufferTime;
    }
    public void SetJumpHeld(bool held)
    {
        isJumpHeld = held;
    }
    public void TakeDamage(float amount)
    {
        currentHealth -= amount;

        if (currentHealth <= 0)
        {
            currentHealth = 0;
            Die();
        }
    }
    private void Die()
    {
        Debug.Log("Player has died!");
        // [VI] Xử lý chết (Reload scene, hiện UI...) / [EN] Handle death (Reload scene, show UI...)
    }




    private void HandleUprightForce()
    {
        Quaternion characterCurrent = transform.rotation;
        Quaternion toGoal = ShortestDistance(targetRotation, characterCurrent);
        toGoal.ToAngleAxis(out float rotDegrees, out Vector3 rotAxis);
        rotAxis.Normalize();

        float rotRadians = rotDegrees * Mathf.Deg2Rad;

        Vector3 torque = (rotAxis * (rotRadians * uprightSpringStrength)) - (rb.angularVelocity * uprightSpringDamper);

        rb.AddTorque(torque);
    }
    private Quaternion ShortestDistance(Quaternion target, Quaternion current)
    {
        //Tính trục y
        float y = (Mathf.Atan2(target.y, current.y) * Mathf.Deg2Rad - 90f);
        Quaternion goal = target * Quaternion.Inverse(current);

        return Quaternion.Euler(goal.eulerAngles.x, y, goal.eulerAngles.z);
    }    
}

﻿using UnityEngine;
using UnityEngine.InputSystem;

[RequireComponent(typeof(PlayerInput))]
public class FrogInput : MonoBehaviour
{
    [Header("Reference")]
    private PlayerInput inputActions;
    private FrogController player;
    private FrogCamera playerCamera;
    private FrogTongue frogTongue;
    private WaterHose waterHose;

    private InputAction moveAction;
    private Vector2 moveValue;

    private InputAction jumpAction;

    private InputAction lookAction;

    private InputAction tongueAction; 


    public bool canInput { get; private set; } = true;

    private void Start()
    {
        // [VI] Khóa chuột / [EN] Lock cursor
        Cursor.lockState = CursorLockMode.Locked;
        player ??= GetComponent<FrogController>();
        playerCamera = FindFirstObjectByType<FrogCamera>();
        frogTongue = GetComponent<FrogTongue>();
        waterHose = GetComponent<WaterHose>(); // Link script
    }

    private void Awake()
    {
        inputActions = GetComponent<PlayerInput>();

        var map = inputActions.actions.FindActionMap("Player");

        // [VI] Tìm các Action / [EN] Find Actions
        moveAction = map.FindAction("Move", true);
        jumpAction = map.FindAction("Jump", true);
        lookAction = map.FindAction("Look", true);
        tongueAction = map.FindAction("Attack", true);

        // [VI] Đăng ký sự kiện khi bấm nút nhảy (cho Jump Buffer) 
        // [EN] Register event when jump button is pressed (for Jump Buffer)
        jumpAction.started += ctx => GetJumpDownInput();
        tongueAction.started += ctx => GetTongueInput();
    }

    private void OnEnable()
    {
        inputActions.actions.FindActionMap("Player").Enable();
    }

    private void OnDisable()
    {
        inputActions.actions.FindActionMap("Player").Disable();
    }

    private void OnDestroy()
    {
        // [VI] Hủy đăng ký an toàn / [EN] Safe unregistration
        if (jumpAction != null)
        {
            jumpAction.started -= ctx => GetJumpDownInput();
        }
    }

    void Update()
    {
        if (!canInput) return;

        GetMoveInput();
        GetLookInput();
        GetJumpHeldInput();
        GetSprayInput();
    }

    // =========================================================
    // [VI] CÁC HÀM XỬ LÝ CHUYÊN BIỆT / [EN] DEDICATED PROCESSING FUNCTIONS
    // =========================================================
    private void GetLookInput()
    {
        // Gửi dữ liệu Vector2 (X, Y) từ chuột/tay cầm sang Camera
        if (playerCamera != null)
        {
            playerCamera.SetLookInput(lookAction.ReadValue<Vector2>());
        }
    }

    private void GetMoveInput()
    {
        moveValue = moveAction.ReadValue<Vector2>();

        if (player != null)
        {
            player.SetMoveInput(moveValue);
        }
    }

    private void GetJumpHeldInput()
    {
        // [VI] Kiểm tra xem nút Jump có đang bị giữ không (Analog Jump)
        // [EN] Check if Jump button is currently being held (Analog Jump)
        bool isHeld = jumpAction.ReadValue<float>() > 0.5f;

        if (player != null)
        {
            player.SetJumpHeld(isHeld);
        }
    }

    private void GetJumpDownInput()
    {
        if (!canInput) return;

        if (player != null)
        {
            // [VI] Kích hoạt lệnh nhảy tức thời / [EN] Trigger immediate jump command
            player.TriggerJumpDown();
        }
    }
    private void GetTongueInput()
    {
        if (!canInput) return;

        // [VI] Gọi hàm bắn/thu lưỡi của FrogTongue / [EN] Call shoot/retract function of FrogTongue
        if (frogTongue != null)
        {
            frogTongue.HandleTongueInput();
        }
    }

    private void GetSprayInput()
    {
        if (waterHose != null)
        {
            // [VI] Kiểm tra nút Attack có đang bị giữ không / [EN] Check if Attack button is held down
            bool isHoldingSpray = tongueAction.ReadValue<float>() > 0.5f;
            waterHose.HandleWaterInput(isHoldingSpray);
        }
    }
}

﻿using UnityEngine;

[RequireComponent(typeof(LineRenderer))]
public class FrogTongue : MonoBehaviour
{
    // [VI] 5 Trạng thái của lưỡi / [EN] 5 States of the tongue
    public enum TongueState { Idle, Extending, Attached, ZippingPlayer, PullingObject, Retracting }
    public TongueState currentState = TongueState.Idle;

    [Header("Tongue Settings")]
    [SerializeField] private Transform mouthPoint;
    [SerializeField] private float maxDistance = 15f;
    [SerializeField] private float shootSpeed = 50f;
    [SerializeField] private float zipSpeed = 25f;
    [SerializeField] private float pullSpeed = 15f;
    [SerializeField] private float tongueRadius = 0.5f;
    [SerializeField] private LayerMask hitLayer;

    [Header("Stomach Settings (Nuốt & Nhổ)")]
    [SerializeField] private float spitForce = 30f; // [VI] Lực phun đồ vật / [EN] Force to spit object
    private GameObject swallowedObject = null;      // [VI] Dạ dày lưu trữ / [EN] Stomach storage

    [Header("Joint Settings")]
    [SerializeField] private float springForce = 100f;
    [SerializeField] private float springDamper = 10f;

    [Header("Safety Settings")]
    [SerializeField] private float maxZipTime = 2.0f;
    [SerializeField] private float pullStopDistance = 1.5f;

    private float zipTimer = 0f;
    private LineRenderer line;
    private Rigidbody rb;
    private FrogController controller;
    private SpringJoint currentJoint;

    private Vector3 targetPoint;
    private Vector3 currentTipPosition;
    private GameObject hitObject;
    private Rigidbody hitRb;

    private void Awake()
    {
        line = GetComponent<LineRenderer>();
        rb = GetComponent<Rigidbody>();
        controller = GetComponent<FrogController>();

        line.positionCount = 2;
        line.enabled = false;
    }

    // =========================================================
    // [VI] GOM CHUNG LOGIC VÀO 1 NÚT BẤM (NHƯ BẠN YÊU CẦU)
    // [EN] COMBINED LOGIC INTO 1 BUTTON PRESS (AS REQUESTED)
    // =========================================================
    public void HandleTongueInput()
    {
        // [VI] Nếu đang ngậm đồ và lưỡi đang rảnh rỗi -> Phun ra
        // [EN] If holding an object and tongue is idle -> Spit it out
        if (swallowedObject != null && currentState == TongueState.Idle)
        {
            SpitObject();
            return; // [VI] Ngắt luôn, không chạy các lệnh bên dưới / [EN] Stop here, don't run below commands
        }

        switch (currentState)
        {
            case TongueState.Idle:
                ShootTongue();
                break;
            case TongueState.Attached:
                StartRetraction();
                break;
            case TongueState.ZippingPlayer:
            case TongueState.PullingObject:
                StopEverything();
                break;
        }
    }

    private void SpitObject()
    {
        swallowedObject.SetActive(true);
        swallowedObject.transform.position = mouthPoint.position + transform.forward * 1.5f;

        Rigidbody spitRb = swallowedObject.GetComponent<Rigidbody>();
        if (spitRb != null)
        {
            spitRb.linearVelocity = Vector3.zero;
            spitRb.AddForce(transform.forward * spitForce * spitRb.mass, ForceMode.Impulse);
        }

        swallowedObject = null; // [VI] Làm rỗng bụng / [EN] Empty stomach
    }

    private void ShootTongue()
    {
        Ray ray = new Ray(mouthPoint.position, transform.forward);
        if (Physics.SphereCast(ray, tongueRadius, out RaycastHit hit, maxDistance, hitLayer))
        {
            targetPoint = hit.point;
            hitObject = hit.collider.gameObject;
            hitRb = hit.collider.GetComponent<Rigidbody>();
        }
        else
        {
            targetPoint = mouthPoint.position + transform.forward * maxDistance;
            hitObject = null;
        }

        currentTipPosition = mouthPoint.position;
        line.enabled = true;
        currentState = TongueState.Extending;
    }

    private void StartRetraction()
    {
        if (currentJoint != null) Destroy(currentJoint);

        if (hitObject != null && hitObject.CompareTag("Grappleable"))
        {
            if (hitRb == null)
            {
                currentState = TongueState.ZippingPlayer;
                controller.isGrappling = true;
                rb.useGravity = false;
                zipTimer = 0f;
            }
            else
            {
                currentState = TongueState.PullingObject;
                hitRb.useGravity = false;
            }
        }
        else
        {
            currentState = TongueState.Retracting;
        }
    }

    private void Update()
    {
        switch (currentState)
        {
            case TongueState.Extending:
                currentTipPosition = Vector3.MoveTowards(currentTipPosition, targetPoint, shootSpeed * Time.deltaTime);
                if (Vector3.Distance(currentTipPosition, targetPoint) < 0.1f) OnHitTarget();
                break;

            case TongueState.Retracting:
                currentTipPosition = Vector3.MoveTowards(currentTipPosition, mouthPoint.position, shootSpeed * 1.5f * Time.deltaTime);
                if (Vector3.Distance(currentTipPosition, mouthPoint.position) < 0.1f) ResetToIdle();
                break;

            case TongueState.Attached:
            case TongueState.ZippingPlayer:
            case TongueState.PullingObject:
                if (hitObject != null)
                {
                    currentTipPosition = (hitRb != null) ? hitRb.position : targetPoint;
                }
                break;
        }
    }

    private void FixedUpdate()
    {
        if (currentState == TongueState.ZippingPlayer)
        {
            zipTimer += Time.fixedDeltaTime;

            Vector3 dir = (targetPoint - transform.position).normalized;
            rb.linearVelocity = dir * zipSpeed;

            if (Vector3.Distance(transform.position, targetPoint) < 1.5f || zipTimer >= maxZipTime)
            {
                StopEverything();
            }
        }
        else if (currentState == TongueState.PullingObject)
        {
            if (hitRb == null) { StopEverything(); return; }

            Vector3 dir = (mouthPoint.position - hitRb.position).normalized;
            hitRb.linearVelocity = dir * pullSpeed;

            Collider[] hitColliders = Physics.OverlapSphere(mouthPoint.position, pullStopDistance, hitLayer);
            foreach (Collider col in hitColliders)
            {
                if (col.gameObject == hitObject)
                {
                    swallowedObject = hitObject;

                    hitRb.linearVelocity = Vector3.zero;
                    hitObject.SetActive(false);
                    StopEverything();
                    break;
                }
            }
        }
    }

    private void LateUpdate()
    {
        if (line.enabled)
        {
            line.SetPosition(0, mouthPoint.position);
            line.SetPosition(1, currentTipPosition);
        }
    }

    private void OnHitTarget()
    {
        if (hitObject == null) { currentState = TongueState.Retracting; return; }

        if (hitObject.CompareTag("Grappleable"))
        {
            currentJoint = gameObject.AddComponent<SpringJoint>();
            currentJoint.autoConfigureConnectedAnchor = false;
            currentJoint.connectedAnchor = targetPoint;
            if (hitRb != null) currentJoint.connectedBody = hitRb;

            currentJoint.spring = springForce;
            currentJoint.damper = springDamper;
            currentJoint.maxDistance = Vector3.Distance(transform.position, targetPoint) * 0.9f;

            currentState = TongueState.Attached;
        }
        else if (hitObject.CompareTag("FrogInteractable"))
        {
            IFrogInteractable interact = hitObject.GetComponent<IFrogInteractable>();
            if (interact != null) interact.OnTongueHit();
            currentState = TongueState.Retracting;
        }
        else
        {
            currentState = TongueState.Retracting;
        }
    }

    private void StopEverything()
    {
        if (hitRb != null) hitRb.useGravity = true;
        controller.isGrappling = false;
        rb.useGravity = true;
        currentState = TongueState.Retracting;
    }

    private void ResetToIdle()
    {
        line.enabled = false;
        currentState = TongueState.Idle;
    }
}

﻿using UnityEngine;

public class FireSource : MonoBehaviour
{
    public float health = 100f;
    public float damageToPlayerPerSecond = 15f;
    public ParticleSystem fireVFX;

    private bool isExtinguished = false;

    public void TakeDamage(float amount)
    {
        if (isExtinguished) return;

        health -= amount;

        if (health <= 0)
        {
            isExtinguished = true;
            if (fireVFX != null) fireVFX.Stop();

            GetComponent<Collider>().enabled = false;
            Destroy(gameObject, 3f); // [VI] Đợi 3s cho hạt lửa bay hết rồi xóa / [EN] Wait 3s for particles to clear then destroy
        }
    }

    private void OnTriggerStay(Collider other)
    {
        if (isExtinguished) return;

        FrogController player = other.GetComponent<FrogController>();
        if (player != null)
        {
            Debug.Log("Damage Player");
            player.TakeDamage(damageToPlayerPerSecond * Time.fixedDeltaTime);
        }
    }
}

﻿using UnityEngine;

public class WaterHose : MonoBehaviour
{
    [Header("Settings")]
    public float damagePerSecond = 30f;
    public float range = 10f;
    public float streamRadius = 0.5f;
    public LayerMask fireLayer;
    public ParticleSystem waterParticles;

    private bool isSpraying = false;

    // [VI] Gọi hàm này từ FrogInput / [EN] Call this from FrogInput
    public void HandleWaterInput(bool isHolding)
    {
        isSpraying = isHolding;

        if (isSpraying && !waterParticles.isPlaying) waterParticles.Play();
        else if (!isSpraying && waterParticles.isPlaying) waterParticles.Stop();
    }

    private void FixedUpdate()
    {
        if (!isSpraying) return;

        Ray ray = new Ray(transform.position, transform.forward);
        if (Physics.SphereCast(ray, streamRadius, out RaycastHit hit, range, fireLayer))
        {
            FireSource fire = hit.collider.GetComponent<FireSource>();
            if (fire != null)
            {
                fire.TakeDamage(damagePerSecond * Time.fixedDeltaTime);
            }
        }
    }
}

    static string s_ReadmeSourceDirectory = "Assets/TutorialInfo";

    const float k_Space = 16f;

    static ReadmeEditor()
    {
        EditorApplication.delayCall += SelectReadmeAutomatically;
    }

    static void RemoveTutorial()
    {
        if (EditorUtility.DisplayDialog("Remove Readme Assets",
            
            $"All contents under {s_ReadmeSourceDirectory} will be removed, are you sure you want to proceed?",
            "Proceed",
            "Cancel"))
        {
            if (Directory.Exists(s_ReadmeSourceDirectory))
            {
                FileUtil.DeleteFileOrDirectory(s_ReadmeSourceDirectory);
                FileUtil.DeleteFileOrDirectory(s_ReadmeSourceDirectory + ".meta");
            }
            else
            {
                Debug.Log($"Could not find the Readme folder at {s_ReadmeSourceDirectory}");
            }

            var readmeAsset = SelectReadme();
            if (readmeAsset != null)
            {
                var path = AssetDatabase.GetAssetPath(readmeAsset);
                FileUtil.DeleteFileOrDirectory(path + ".meta");
                FileUtil.DeleteFileOrDirectory(path);
            }

            AssetDatabase.Refresh();
        }
    }

    static void SelectReadmeAutomatically()
    {
        if (!SessionState.GetBool(s_ShowedReadmeSessionStateName, false))
        {
            var readme = SelectReadme();
            SessionState.SetBool(s_ShowedReadmeSessionStateName, true);

            if (readme && !readme.loadedLayout)
            {
                LoadLayout();
                readme.loadedLayout = true;
            }
        }
    }

    static void LoadLayout()
    {
        var assembly = typeof(EditorApplication).Assembly;
        var windowLayoutType = assembly.GetType("UnityEditor.WindowLayout", true);
        var method = windowLayoutType.GetMethod("LoadWindowLayout", BindingFlags.Public | BindingFlags.Static);
        method.Invoke(null, new object[] { Path.Combine(Application.dataPath, "TutorialInfo/Layout.wlt"), false });
    }

    static Readme SelectReadme()
    {
        var ids = AssetDatabase.FindAssets("Readme t:Readme");
        if (ids.Length == 1)
        {
            var readmeObject = AssetDatabase.LoadMainAssetAtPath(AssetDatabase.GUIDToAssetPath(ids[0]));

            Selection.objects = new UnityEngine.Object[] { readmeObject };

            return (Readme)readmeObject;
        }
        else
        {
            Debug.Log("Couldn't find a readme");
            return null;
        }
    }

    protected override void OnHeaderGUI()
    {
        var readme = (Readme)target;
        Init();

        var iconWidth = Mathf.Min(EditorGUIUtility.currentViewWidth / 3f - 20f, 128f);

        GUILayout.BeginHorizontal("In BigTitle");
        {
            if (readme.icon != null)
            {
                GUILayout.Space(k_Space);
                GUILayout.Label(readme.icon, GUILayout.Width(iconWidth), GUILayout.Height(iconWidth));
            }
            GUILayout.Space(k_Space);
            GUILayout.BeginVertical();
            {

                GUILayout.FlexibleSpace();
                GUILayout.Label(readme.title, TitleStyle);
                GUILayout.FlexibleSpace();
            }
            GUILayout.EndVertical();
            GUILayout.FlexibleSpace();
        }
        GUILayout.EndHorizontal();
    }

    public override void OnInspectorGUI()
    {
        var readme = (Readme)target;
        Init();

        foreach (var section in readme.sections)
        {
            if (!string.IsNullOrEmpty(section.heading))
            {
                GUILayout.Label(section.heading, HeadingStyle);
            }

            if (!string.IsNullOrEmpty(section.text))
            {
                GUILayout.Label(section.text, BodyStyle);
            }

            if (!string.IsNullOrEmpty(section.linkText))
            {
                if (LinkLabel(new GUIContent(section.linkText)))
                {
                    Application.OpenURL(section.url);
                }
            }

            GUILayout.Space(k_Space);
        }

        if (GUILayout.Button("Remove Readme Assets", ButtonStyle))
        {
            RemoveTutorial();
        }
    }

    bool m_Initialized;

    GUIStyle LinkStyle
    {
        get { return m_LinkStyle; }
    }

    [SerializeField]
    GUIStyle m_LinkStyle;

    GUIStyle TitleStyle
    {
        get { return m_TitleStyle; }
    }

    [SerializeField]
    GUIStyle m_TitleStyle;

    GUIStyle HeadingStyle
    {
        get { return m_HeadingStyle; }
    }

    [SerializeField]
    GUIStyle m_HeadingStyle;

    GUIStyle BodyStyle
    {
        get { return m_BodyStyle; }
    }

    [SerializeField]
    GUIStyle m_BodyStyle;

    GUIStyle ButtonStyle
    {
        get { return m_ButtonStyle; }
    }

    [SerializeField]
    GUIStyle m_ButtonStyle;

    void Init()
    {
        if (m_Initialized)
            return;
        m_BodyStyle = new GUIStyle(EditorStyles.label);
        m_BodyStyle.wordWrap = true;
        m_BodyStyle.fontSize = 14;
        m_BodyStyle.richText = true;

        m_TitleStyle = new GUIStyle(m_BodyStyle);
        m_TitleStyle.fontSize = 26;

        m_HeadingStyle = new GUIStyle(m_BodyStyle);
        m_HeadingStyle.fontStyle = FontStyle.Bold;
        m_HeadingStyle.fontSize = 18;

        m_LinkStyle = new GUIStyle(m_BodyStyle);
        m_LinkStyle.wordWrap = false;

        // Match selection color which works nicely for both light and dark skins
        m_LinkStyle.normal.textColor = new Color(0x00 / 255f, 0x78 / 255f, 0xDA / 255f, 1f);
        m_LinkStyle.stretchWidth = false;

        m_ButtonStyle = new GUIStyle(EditorStyles.miniButton);
        m_ButtonStyle.fontStyle = FontStyle.Bold;

        m_Initialized = true;
    }

    bool LinkLabel(GUIContent label, params GUILayoutOption[] options)
    {
        var position = GUILayoutUtility.GetRect(label, LinkStyle, options);

        Handles.BeginGUI();
        Handles.color = LinkStyle.normal.textColor;
        Handles.DrawLine(new Vector3(position.xMin, position.yMax), new Vector3(position.xMax, position.yMax));
        Handles.color = Color.white;
        Handles.EndGUI();

        EditorGUIUtility.AddCursorRect(position, MouseCursor.Link);

        return GUI.Button(position, label, LinkStyle);
    }
}

﻿using UnityEngine;
using UnityEngine.EventSystems;
using DG.Tweening;

// [VI] Thêm ISelectHandler và IDeselectHandler để nhận tín hiệu từ Gamepad
// [EN] Add ISelectHandler and IDeselectHandler to receive Gamepad signals
public class Uicontrol : MonoBehaviour,
    IPointerDownHandler, IPointerUpHandler,
    IPointerEnterHandler, IPointerExitHandler,
    ISelectHandler, IDeselectHandler // <--- MỚI / NEW
{
    [Header("Animation Settings")]
    [SerializeField] private float animDuration = 0.2f;
    [SerializeField] private float hoverScaleMultiplier = 1.2f;

    private Vector3 normalScale;
    private bool isSelected;

    void Awake()
    {
        normalScale = transform.localScale;
    }

    // --- LOGIC CHUNG CHO CẢ CHUỘT VÀ TAY CẦM / COMMON LOGIC FOR MOUSE & GAMEPAD ---

    private void PlayEnterAnimation()
    {
        if (isSelected) return;
        isSelected = true;
        transform.DOKill();
        transform.localRotation = Quaternion.identity;
        transform.DOScale(normalScale * hoverScaleMultiplier, animDuration).SetEase(Ease.OutBack);
        transform.DOPunchRotation(new Vector3(0, 0, 5), 0.15f, 20, 1);
    }

    private void PlayExitAnimation()
    {
        if (!isSelected) return;
        isSelected = false;
        transform.DOKill();
        transform.localRotation = Quaternion.identity;
        transform.DOScale(normalScale, animDuration).SetEase(Ease.OutBack);
    }

    // --- INTERFACES ---

    // [VI] Khi dùng chuột di qua / [EN] When mouse hovers
    public void OnPointerEnter(PointerEventData eventData) => PlayEnterAnimation();
    public void OnPointerExit(PointerEventData eventData) => PlayExitAnimation();

    // [VI] Khi dùng Gamepad chọn nút / [EN] When Gamepad selects the button
    public void OnSelect(BaseEventData eventData) => PlayEnterAnimation();
    public void OnDeselect(BaseEventData eventData) => PlayExitAnimation();

    public void OnPointerDown(PointerEventData eventData)
    {
        transform.DOKill();
        transform.DOScale(normalScale, animDuration).SetEase(Ease.OutBack);
    }

    public void OnPointerUp(PointerEventData eventData)
    {
        transform.DOKill();
        transform.DOScale(normalScale * hoverScaleMultiplier, animDuration).SetEase(Ease.OutBack);
    }
}
